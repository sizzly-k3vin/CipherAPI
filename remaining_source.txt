struct CAPIClient {
    std::string host;
    uint16_t port;
    std::unique_ptr<LibraryManager> libraries;
    std::unique_ptr<CipherVideo::Receiver> video_receiver;
    int fd = -1;
    std::mutex send_mutex;
    std::mutex receive_mutex;

    CAPIClient(const char* h, uint16_t p, const char* directory)
        : host(h), port(p),
          libraries(std::make_unique<LibraryManager>(directory)) {
        video_receiver =
            std::make_unique<CipherVideo::Receiver>(*libraries);
    }
};

static thread_local std::string last_error;

static void set_error(const char* message) {
    last_error = message ? message : "Unknown CipherAPI error";
}

static int wait_readable(int fd, int timeout_ms) {
    pollfd descriptor{fd, POLLIN, 0};
    int result;

    do {
        result = poll(&descriptor, 1, timeout_ms);
    } while (result < 0 && errno == EINTR);

    if (result < 0)
        throw std::runtime_error("Socket poll failed");

    return result;
}

extern "C" {

const char* capi_last_error(void) {
    return last_error.c_str();
}

CAPIClient* capi_client_create(
    const char* host, uint16_t port, const char* library_directory
) {
    try {
        if (!host || !*host || !library_directory || !*library_directory)
            throw std::invalid_argument("Host and library directory are required");

        last_error.clear();
        return new CAPIClient(host, port, library_directory);
    } catch (const std::exception& e) {
        set_error(e.what());
        return nullptr;
    } catch (...) {
        set_error("Unknown error creating CipherAPI client");
        return nullptr;
    }
}

int capi_client_connect(CAPIClient* client) {
    try {
        if (!client)
            throw std::invalid_argument("Invalid client handle");

        if (client->fd >= 0)
            return 0;

        addrinfo hints{};
        hints.ai_family = AF_UNSPEC;
        hints.ai_socktype = SOCK_STREAM;

        addrinfo* result = nullptr;
        const std::string service = std::to_string(client->port);

        int status = getaddrinfo(
            client->host.c_str(), service.c_str(), &hints, &result
        );

        if (status != 0)
            throw std::runtime_error(gai_strerror(status));

        int connected_fd = -1;

        for (addrinfo* item = result; item; item = item->ai_next) {
            int candidate = socket(
                item->ai_family, item->ai_socktype, item->ai_protocol
            );

            if (candidate < 0)
                continue;

            if (::connect(candidate, item->ai_addr, item->ai_addrlen) == 0) {
                connected_fd = candidate;
                break;
            }

            ::close(candidate);
        }

        freeaddrinfo(result);

        if (connected_fd < 0)
            throw std::runtime_error("Could not connect to CipherAPI server");

        client->fd = connected_fd;
        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown connection error");
        return -1;
    }
}

int capi_client_send_text(CAPIClient* client, const char* message, int fast) {
    try {
        if (!client || !message)
            throw std::invalid_argument("Invalid client or message");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->send_mutex);
        std::string text(message);

        if (fast) {
            CipherTransport::send_frame(
                client->fd, CipherTransport::FAST_TEXT, text
            );
        } else {
            std::string encoded = client->libraries->encode(text);
            CipherTransport::send_frame(
                client->fd, CipherTransport::TEXT, encoded
            );
        }

        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown text send error");
        return -1;
    }
}

int capi_client_send_image(CAPIClient* client, const char* path, int raw) {
    try {
        if (!client || !path || !*path)
            throw std::invalid_argument("Invalid image path");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->send_mutex);

        std::string packet = raw
            ? ImageLibrary::pack_raw(path)
            : ImageLibrary::pack(path);

        auto type = raw
            ? CipherTransport::RAW_IMAGE
            : CipherTransport::IMAGE;

        CipherTransport::send_frame(client->fd, type, packet);

        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown image send error");
        return -1;
    }
}

int capi_client_send_video(CAPIClient* client, const char* path, int checksum) {
    try {
        if (!client || !path || !*path)
            throw std::invalid_argument("Invalid video path");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        CipherVideo::send_file(

        return 0;

        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown video send error");
        return -1;
    }
}

int capi_client_receive_event(
    CAPIClient* client, int* event_type,
    char* buffer, size_t capacity, int timeout_ms
) {
    try {
        if (!client || !event_type || !buffer || capacity == 0)
            throw std::invalid_argument("Invalid receive arguments");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->receive_mutex);

        while (true) {
            int ready = wait_readable(client->fd, timeout_ms);

            if (ready == 0)
                return -2;

            uint8_t type = 0;
            std::string payload;

            if (!CipherTransport::receive_frame(client->fd, type, payload))
                throw std::runtime_error("Server closed the connection");

            int kind = 0;
            std::string output;

            if (type == CipherTransport::TEXT) {
                output = client->libraries->decode(payload);
                kind = 1;
            } else if (type == CipherTransport::FAST_TEXT) {
                output = std::move(payload);
                kind = 1;
            } else if (type == CipherTransport::IMAGE) {
                output = ImageLibrary::unpack(
                    payload, "received_images"
                );
                kind = 2;
            } else if (type == CipherTransport::RAW_IMAGE) {
                output = ImageLibrary::unpack_raw(
                    payload, "received_images"
                );
                kind = 2;
            } else if (
                type >= CipherTransport::VIDEO_START &&
                type <= CipherTransport::VIDEO_END
            ) {
                output = client->video_receiver->handle(type, payload);

                if (output.empty())
                    continue;

                kind = 3;
            } else {
                throw std::runtime_error("Unsupported incoming frame type");
            }

            if (output.size() + 1 > capacity)
                throw std::runtime_error("Receive buffer is too small");

            std::memcpy(buffer, output.data(), output.size());
            buffer[output.size()] = 0;
            *event_type = kind;

            last_error.clear();
            return static_cast<int>(output.size());
        }
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown receive error");
        return -1;
    }
}

int capi_client_receive(
    CAPIClient* client, char* buffer, size_t capacity, int timeout_ms
) {
    int event_type = 0;

    int result = capi_client_receive_event(
        client, &event_type, buffer, capacity, timeout_ms
    );

    if (result < 0)
        return result;

    if (event_type != 1) {
        set_error("Received media event; use receive_event() instead");
        return -1;
    }

    return result;
}

void capi_client_close(CAPIClient* client) {
    if (!client || client->fd < 0)
        return;

    shutdown(client->fd, SHUT_RDWR);
    ::close(client->fd);
    client->fd = -1;
}

void capi_client_destroy(CAPIClient* client) {
    if (!client)
        return;

    capi_client_close(client);
    delete client;
}

constexpr std::uint8_t VERSION = 1;
constexpr std::size_t HASH_SIZE = 32;
constexpr std::size_t CHUNK_HEADER_SIZE = 14;

void put_u16(std::string& out, std::uint16_t v) {
    out.push_back(static_cast<char>(v >> 8));
    out.push_back(static_cast<char>(v));
}

void put_u32(std::string& out, std::uint32_t v) {
    for (int i = 3; i >= 0; --i)
        out.push_back(static_cast<char>(v >> (i * 8)));
}

void put_u64(std::string& out, std::uint64_t v) {
    for (int i = 7; i >= 0; --i)
        out.push_back(static_cast<char>(v >> (i * 8)));
}

std::uint16_t get_u16(const std::string& s, std::size_t p) {
    return (static_cast<std::uint16_t>(
                static_cast<unsigned char>(s[p])) << 8) |
           static_cast<unsigned char>(s[p + 1]);
}

std::uint32_t get_u32(const std::string& s, std::size_t p) {
    std::uint32_t v = 0;
    for (int i = 0; i < 4; ++i)
        v = (v << 8) | static_cast<unsigned char>(s[p + i]);
    return v;
}

std::uint64_t get_u64(const std::string& s, std::size_t p) {
    std::uint64_t v = 0;
    for (int i = 0; i < 8; ++i)
        v = (v << 8) | static_cast<unsigned char>(s[p + i]);
    return v;
}

// A deterministic generator: both ends produce the same permutation.
class SplitMix64 {
public:
    explicit SplitMix64(std::uint64_t seed) : state_(seed) {}

    std::uint64_t next() {
        std::uint64_t z = (state_ += 0x9e3779b97f4a7c15ULL);
        z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;
        z = (z ^ (z >> 27)) * 0x94d049bb133111ebULL;
        return z ^ (z >> 31);
    }

private:
    std::uint64_t state_;
};

std::vector<std::uint32_t> make_permutation(
    std::size_t count, std::uint64_t seed) {

    std::vector<std::uint32_t> permutation(count);
    std::iota(permutation.begin(), permutation.end(), 0);

    SplitMix64 rng(seed);

    for (std::size_t i = count; i > 1; --i) {
        const std::size_t j =
            static_cast<std::size_t>(rng.next() % i);
        std::swap(permutation[i - 1], permutation[j]);
    }

    return permutation;
}

std::string bytes_to_bits(const std::string& bytes) {
    std::string bits;
    bits.reserve(bytes.size() * 8);

    for (unsigned char byte : bytes) {
        for (int bit = 7; bit >= 0; --bit)
            bits.push_back((byte & (1u << bit)) ? '1' : '0');
    }

    return bits;
}

std::string bits_to_bytes(const std::string& bits) {
    if (bits.size() % 8 != 0)
        throw std::runtime_error("Invalid reconstructed bit count");

    std::string bytes(bits.size() / 8, '\0');

    for (std::size_t i = 0; i < bits.size(); ++i) {
        if (bits[i] != '0' && bits[i] != '1')
            throw std::runtime_error("Invalid bit in shuffled data");

        if (bits[i] == '1')
            bytes[i / 8] |= static_cast<char>(1u << (7 - (i % 8)));
    }

    return bytes;
}

std::string shuffle_bits(
    const std::string& original, std::uint64_t seed) {

    const auto permutation =
        make_permutation(original.size(), seed);

    std::string shuffled(original.size(), '0');

    for (std::size_t i = 0; i < original.size(); ++i)
        shuffled[i] = original[permutation[i]];

    return shuffled;
}

std::string unshuffle_bits(
    const std::string& shuffled, std::uint64_t seed) {

    const auto permutation =
        make_permutation(shuffled.size(), seed);

    std::string original(shuffled.size(), '0');

    for (std::size_t i = 0; i < shuffled.size(); ++i)
        original[permutation[i]] = shuffled[i];

    return original;
}

void digest_update(EVP_MD_CTX* ctx, const std::string& bytes) {
    if (EVP_DigestUpdate(ctx, bytes.data(), bytes.size()) != 1)
        throw std::runtime_error("SHA-256 update failed");
}

std::string digest_final(EVP_MD_CTX* ctx) {
    std::string digest(HASH_SIZE, '\0');
    unsigned int length = 0;

    if (EVP_DigestFinal_ex(
            ctx,
            reinterpret_cast<unsigned char*>(digest.data()),
            &length) != 1 || length != HASH_SIZE) {
        throw std::runtime_error("SHA-256 finalization failed");
    }

    return digest;
}

std::string safe_output_name(const std::string& name) {
    if (name.empty() || name.size() > 255 ||
        name == "." || name == ".." ||
        fs::path(name).filename().string() != name ||
        name.find('/') != std::string::npos ||
        name.find('\\') != std::string::npos) {
        throw std::runtime_error("Unsafe video filename");
    }

    return name;
}

class Receiver::State {
public:
    std::ofstream file;
    EVP_MD_CTX* digest = nullptr;

    ~State() {
        if (digest)
            EVP_MD_CTX_free(digest);
    }
};

TransferStats send_file(
    int fd,
    const LibraryManager& libraries,
    const std::string& path,
    bool use_sha256) {

    const fs::path source(path);
    const std::string filename = source.filename().string();

    if (filename.empty() || filename.size() > 255)
        throw std::runtime_error("Invalid video filename");

    std::error_code ec;
    const std::uint64_t file_size = fs::file_size(source, ec);

    if (ec)
        throw std::runtime_error("Cannot read video file size: " + ec.message());

    if (file_size == 0)
        throw std::runtime_error("Video file is empty");

    std::ifstream input(source, std::ios::binary);
    if (!input)
        throw std::runtime_error("Cannot open video file");

    EVP_MD_CTX* digest = nullptr;

    if (use_sha256) {
        digest = EVP_MD_CTX_new();
        if (!digest)
            throw std::runtime_error("Cannot allocate SHA-256 context");

        if (EVP_DigestInit_ex(digest, EVP_sha256(), nullptr) != 1) {
            EVP_MD_CTX_free(digest);
            throw std::runtime_error("Cannot initialize SHA-256");
        }
    }

    TransferStats stats;
    stats.original_bytes = file_size;

    try {
        // VIDEO_START: version, flags, filename length, file size, filename.
        std::string start;
        start.push_back(static_cast<char>(VERSION));
        start.push_back(static_cast<char>(use_sha256 ? 1 : 0));
        put_u16(start, static_cast<std::uint16_t>(filename.size()));
        put_u64(start, file_size);
        start += filename;

        CipherTransport::send_frame(
            fd, CipherTransport::VIDEO_START, start);
        stats.wire_bytes += start.size();

        std::uint64_t remaining = file_size;
        std::uint32_t sequence = 0;

        while (remaining > 0) {
            const std::size_t amount = static_cast<std::size_t>(
                std::min<std::uint64_t>(CHUNK_SIZE, remaining));

            std::string bytes(amount, '\0');
            input.read(bytes.data(), static_cast<std::streamsize>(amount));

            if (input.gcount() != static_cast<std::streamsize>(amount))
                throw std::runtime_error("Video file changed or read failed");

            if (use_sha256)
                digest_update(digest, bytes);

            std::uint64_t seed = 0;
            if (RAND_bytes(
                    reinterpret_cast<unsigned char*>(&seed),
                    sizeof(seed)) != 1) {
                throw std::runtime_error("Random seed generation failed");
            }

            const std::string bits = bytes_to_bits(bytes);
            const std::string shuffled = shuffle_bits(bits, seed);

            // The existing ten libraries encode the shuffled 0/1 characters.
            const std::string encoded = libraries.encode(shuffled);

            // VIDEO_DATA: sequence, seed, original byte count, encoded bits.
            std::string packet;
            packet.reserve(CHUNK_HEADER_SIZE + encoded.size());

            put_u32(packet, sequence);
            put_u64(packet, seed);
            put_u16(packet, static_cast<std::uint16_t>(amount));
            packet += encoded;

            CipherTransport::send_frame(
                fd, CipherTransport::VIDEO_DATA, packet);

            stats.wire_bytes += packet.size();
            ++stats.chunks;
            ++sequence;
            remaining -= amount;
        }

        if (input.peek() != std::char_traits<char>::eof())
            throw std::runtime_error("Video file grew during transfer");

        // VIDEO_END: total original bytes, chunk count, optional digest.
        std::string end;
        put_u64(end, file_size);
        put_u32(end, stats.chunks);

        if (use_sha256)
            end += digest_final(digest);

        CipherTransport::send_frame(
            fd, CipherTransport::VIDEO_END, end);
        stats.wire_bytes += end.size();

        if (digest)
            EVP_MD_CTX_free(digest);

        return stats;
    } catch (...) {
        if (digest)
            EVP_MD_CTX_free(digest);
        throw;
    }
}

Receiver::Receiver(const LibraryManager& libraries)
    : libraries_(libraries), state_(new State()) {}

Receiver::~Receiver() {
    abort_transfer();
    delete state_;
}

void Receiver::abort_transfer() noexcept {
    if (state_ && state_->file.is_open())
        state_->file.close();

    if (!temporary_path_.empty()) {
        std::error_code ec;
        fs::remove(temporary_path_, ec);
    }

    active_ = false;
    expected_size_ = 0;
    received_size_ = 0;
    expected_sequence_ = 0;
    use_sha256_ = false;
    final_path_.clear();
    temporary_path_.clear();

    if (state_ && state_->digest) {
        EVP_MD_CTX_free(state_->digest);
        state_->digest = nullptr;
    }
}

std::string Receiver::handle(
    std::uint8_t type, const std::string& payload) {

    if (type == CipherTransport::VIDEO_START) {
        if (active_)
            throw std::runtime_error("A video transfer is already active");

        if (payload.size() < 12)
            throw std::runtime_error("Invalid VIDEO_START header");

        if (static_cast<std::uint8_t>(payload[0]) != VERSION)
            throw std::runtime_error("Unsupported CVS version");

        const std::uint8_t flags =
            static_cast<std::uint8_t>(payload[1]);

        if (flags > 1)
            throw std::runtime_error("Invalid video checksum mode");

        const std::uint16_t name_length = get_u16(payload, 2);
        expected_size_ = get_u64(payload, 4);

        if (name_length == 0 || name_length > 255 ||
            payload.size() != 12 + name_length ||
            expected_size_ == 0) {
            throw std::runtime_error("Invalid video metadata");
        }

        const std::string filename =
            safe_output_name(payload.substr(12, name_length));

        use_sha256_ = flags == 1;

        if (use_sha256_) {
            state_->digest = EVP_MD_CTX_new();
            if (!state_->digest ||
                EVP_DigestInit_ex(
                    state_->digest, EVP_sha256(), nullptr) != 1) {
                throw std::runtime_error("Cannot initialize receive SHA-256");
            }
        }

        std::string candidate = "received_" + filename;
        unsigned int suffix = 1;

        while (fs::exists(candidate) || fs::exists(candidate + ".part")) {
            candidate = "received_" + filename + "." +
                        std::to_string(suffix++);
        }

        final_path_ = candidate;
        temporary_path_ = final_path_ + ".part";

        state_->file.open(
            temporary_path_, std::ios::binary | std::ios::out);

        if (!state_->file)
            throw std::runtime_error("Cannot create temporary video file");

        received_size_ = 0;
        expected_sequence_ = 0;
        active_ = true;
        return {};
    }

    if (!active_)
        throw std::runtime_error("Video data received without VIDEO_START");

    if (type == CipherTransport::VIDEO_DATA) {
        if (payload.size() < CHUNK_HEADER_SIZE)
            throw std::runtime_error("Invalid VIDEO_DATA header");

        const std::uint32_t sequence = get_u32(payload, 0);
        const std::uint64_t seed = get_u64(payload, 4);
        const std::uint16_t raw_size = get_u16(payload, 12);

        if (sequence != expected_sequence_)
            throw std::runtime_error("Video chunk sequence mismatch");

        if (raw_size == 0 || raw_size > CHUNK_SIZE ||
            raw_size > expected_size_ - received_size_) {
            throw std::runtime_error("Invalid video chunk size");
        }

        const std::string encoded =
            payload.substr(CHUNK_HEADER_SIZE);

        if (encoded.size() !=
            static_cast<std::size_t>(raw_size) * 8 * 64) {
            throw std::runtime_error("Invalid encoded video chunk length");
        }

        const std::string shuffled = libraries_.decode(encoded);

        if (shuffled.size() != static_cast<std::size_t>(raw_size) * 8)
            throw std::runtime_error("Decoded bit count mismatch");

        const std::string bits = unshuffle_bits(shuffled, seed);
        const std::string bytes = bits_to_bytes(bits);

        if (bytes.size() != raw_size)
            throw std::runtime_error("Reconstructed chunk size mismatch");

        state_->file.write(
            bytes.data(), static_cast<std::streamsize>(bytes.size()));

        if (!state_->file)
            throw std::runtime_error("Video disk write failed");

        if (use_sha256_)
            digest_update(state_->digest, bytes);

        received_size_ += bytes.size();
        ++expected_sequence_;
        return {};
    }

    if (type == CipherTransport::VIDEO_END) {
        const std::size_t expected_end_size =
            use_sha256_ ? 12 + HASH_SIZE : 12;

        if (payload.size() != expected_end_size)
            throw std::runtime_error("Invalid VIDEO_END packet");

        const std::uint64_t final_size = get_u64(payload, 0);
        const std::uint32_t final_chunks = get_u32(payload, 8);

        const std::uint64_t expected_chunks =
            (expected_size_ + CHUNK_SIZE - 1) / CHUNK_SIZE;

        if (final_size != expected_size_ ||
            received_size_ != expected_size_ ||
            final_chunks != expected_sequence_ ||
            final_chunks != expected_chunks) {
            throw std::runtime_error("Video transfer size or chunk mismatch");
        }

        if (use_sha256_) {
            const std::string actual = digest_final(state_->digest);
            const std::string claimed = payload.substr(12, HASH_SIZE);

            if (CRYPTO_memcmp(
                    actual.data(), claimed.data(), HASH_SIZE) != 0) {
                throw std::runtime_error("Video SHA-256 verification failed");
            }
        }

        state_->file.flush();
        if (!state_->file)
            throw std::runtime_error("Video file flush failed");

        state_->file.close();

        std::error_code ec;
        fs::rename(temporary_path_, final_path_, ec);

        if (ec)
            throw std::runtime_error("Cannot finalize video file: " +
                                     ec.message());

        const std::string saved = final_path_;

        active_ = false;
        expected_size_ = 0;
        received_size_ = 0;
        expected_sequence_ = 0;
        use_sha256_ = false;
        final_path_.clear();
        temporary_path_.clear();

        if (state_->digest) {
            EVP_MD_CTX_free(state_->digest);
            state_->digest = nullptr;
        }

        return saved;
    }

    throw std::runtime_error("Unexpected frame during video transfer");
}

#include <netinet/in.h>
#include <sys/socket.h>
#include <sys/poll.h>
#include <unistd.h>

#include <atomic>
#include <chrono>
#include <filesystem>
#include <iostream>
#include <memory>
#include <mutex>
#include <sstream>
#include <string>
#include <thread>
#include <vector>

struct Client {
    int id;
    int fd;
    std::atomic<bool> connected{true};
    std::atomic<bool> fast{false};
    Neon::Stats stats;
    std::mutex send_mutex;
    std::thread receiver;
};

static std::atomic<bool> running{true};
static std::mutex clients_mutex;
static std::vector<std::shared_ptr<Client>> clients;

static void disconnect_client(const std::shared_ptr<Client>& c) {
    if (c->connected.exchange(false))
        shutdown(c->fd, SHUT_RDWR);
}

static void send_text(
    const std::shared_ptr<Client>& c,
    const LibraryManager& libraries,
    const std::string& message
) {
    std::lock_guard<std::mutex> lock(c->send_mutex);

    if (!c->connected)
        throw std::runtime_error("Client disconnected");

    std::string payload = c->fast
        ? message
        : libraries.encode(message);

    auto type = c->fast
        ? CipherTransport::FAST_TEXT
        : CipherTransport::TEXT;

    CipherTransport::send_frame(c->fd, type, payload);

    ++c->stats.sent_messages;
    c->stats.sent_chars += message.size();
    c->stats.sent_bytes += payload.size();

    Neon::print_message("SERVER -> CLIENT " +
                        std::to_string(c->id), message);
}

static void receive_client(
    std::shared_ptr<Client> c,
    const LibraryManager& libraries
) {
    CipherVideo::Receiver video_receiver(libraries);

    try {
        while (running && c->connected) {
            uint8_t type = 0;
            std::string payload;

            if (!CipherTransport::receive_frame(c->fd, type, payload))
                break;

            ++c->stats.received_bytes;
            c->stats.received_bytes += payload.size();

            if (type == CipherTransport::TEXT) {
                std::string message = libraries.decode(payload);
                ++c->stats.received_messages;
                c->stats.received_chars += message.size();
                Neon::decode("CLIENT " + std::to_string(c->id) + ": " + message);
                Neon::print_message("CLIENT " +
                                    std::to_string(c->id), message);

                if (message == "/quit")
                    break;
            }
            else if (type == CipherTransport::FAST_TEXT) {
                ++c->stats.received_messages;
                c->stats.received_chars += payload.size();
                Neon::print_message("CLIENT " +
                                    std::to_string(c->id), payload);

                if (payload == "/quit")
                    break;
            }
            else if (type == CipherTransport::IMAGE) {
                std::filesystem::path dir =
                    "received_images/client_" + std::to_string(c->id);
                std::filesystem::create_directories(dir);
                std::string path =
                    ImageLibrary::unpack(payload, dir.string());
                Neon::info("Image received: " + path);
            }
            else if (type == CipherTransport::RAW_IMAGE) {
                std::filesystem::path dir =
                    "received_images/client_" + std::to_string(c->id);
                std::filesystem::create_directories(dir);
                std::string path =
                    ImageLibrary::unpack_raw(payload, dir.string());
                Neon::info("Raw image received: " + path);
            }
            else if (type >= CipherTransport::VIDEO_START &&
                     type <= CipherTransport::VIDEO_END) {
                std::string result = video_receiver.handle(type, payload);
                if (!result.empty())
                    Neon::info("Video received: " + result);
            }
            else {
                Neon::error("Unknown frame from client " +
                            std::to_string(c->id));
            }
        }
    }
    catch (const std::exception& e) {
        Neon::error("Client " + std::to_string(c->id) +
                    ": " + e.what());
    }

    disconnect_client(c);
    Neon::info("Client " + std::to_string(c->id) + " disconnected");
}

static std::shared_ptr<Client> find_client(int id) {
    std::lock_guard<std::mutex> lock(clients_mutex);

    for (auto& c : clients)
        if (c->id == id && c->connected)
            return c;

    return nullptr;
}

static void cleanup_clients() {
    std::vector<std::shared_ptr<Client>> dead;

    {
        std::lock_guard<std::mutex> lock(clients_mutex);

        for (auto it = clients.begin(); it != clients.end();) {
            if (!(*it)->connected) {
                dead.push_back(*it);
                it = clients.erase(it);
            } else {
                ++it;
            }
        }
    }

    for (auto& c : dead) {
        if (c->receiver.joinable())
            c->receiver.join();

        close(c->fd);
    }
}

static void transmit_image(
    const std::shared_ptr<Client>& c,
    const std::string& path,
    bool raw
) {
    std::lock_guard<std::mutex> lock(c->send_mutex);

    if (!c->connected)
        throw std::runtime_error("Client disconnected");

    std::string packet = raw
        ? ImageLibrary::pack_raw(path)
        : ImageLibrary::pack(path);

    auto type = raw
        ? CipherTransport::RAW_IMAGE
        : CipherTransport::IMAGE;

    CipherTransport::send_frame(c->fd, type, packet);

    ++c->stats.sent_messages;
    c->stats.sent_bytes += packet.size();
}

static void transmit_video(
    const std::shared_ptr<Client>& c,
    const LibraryManager& libraries,
    const std::string& path,
    bool sha
) {
    std::lock_guard<std::mutex> lock(c->send_mutex);

    if (!c->connected)
        throw std::runtime_error("Client disconnected");

    CipherVideo::send_file(c->fd, libraries, path, sha);
}

static void show_clients() {
    std::lock_guard<std::mutex> lock(clients_mutex);

    std::cout << "\n--- ACTIVE CLIENTS ---\n";

    for (const auto& c : clients) {
        if (c->connected)
            std::cout << "ID " << c->id
                      << " | " << (c->fast ? "FAST" : "ENCODED")
                      << '\n';
    }

    std::cout << "----------------------\n";
}

int main() {
    try {
        LibraryManager libraries("libraries");

        int server = socket(AF_INET, SOCK_STREAM, 0);
        if (server < 0)
            throw std::runtime_error("Socket creation failed");

        int opt = 1;
        setsockopt(server, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

        sockaddr_in address{};
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;
        address.sin_port = htons(8080);

        if (bind(server, reinterpret_cast<sockaddr*>(&address),
                 sizeof(address)) < 0)
            throw std::runtime_error("Bind failed");

        if (listen(server, 20) < 0)
            throw std::runtime_error("Listen failed");

        Neon::banner("SERVER");
        Neon::info("CipherAPI multi-client server started on port 8080");

        std::thread accept_thread([&]() {
            int next_id = 1;

            while (running) {
                pollfd pfd{server, POLLIN, 0};
                int ready = poll(&pfd, 1, 500);

                if (!running)
                    break;

                if (ready <= 0)
                    continue;

                sockaddr_in client_addr{};
                socklen_t len = sizeof(client_addr);

                int fd = accept(server,
                    reinterpret_cast<sockaddr*>(&client_addr), &len);

                if (fd < 0)
                    continue;

                auto c = std::make_shared<Client>();
                c->id = next_id++;
                c->fd = fd;

                c->receiver = std::thread(
                    receive_client, c, std::cref(libraries));

                {
                    std::lock_guard<std::mutex> lock(clients_mutex);
                    clients.push_back(c);
                }

                Neon::info("Client " + std::to_string(c->id) +
                           " connected from " +
                           std::string(inet_ntoa(client_addr.sin_addr)));
            }
        });

        int selected_id = -1;
        std::string line;

        Neon::info("Type /help for commands");

        while (running) {
            cleanup_clients();

            std::cout << "CipherAPI";
            if (selected_id != -1)
                std::cout << "[" << selected_id << "]";
            std::cout << "> " << std::flush;

            if (!std::getline(std::cin, line))
                break;

            if (line.empty())
                continue;

            std::istringstream input(line);
            std::string command;
            input >> command;

            try {
                if (command == "/help") {
                    std::cout
                        << "\n/help                 Show commands\n"
                        << "/clients              List clients\n"
                        << "/use ID               Select client\n"
                        << "/to ID MESSAGE        Direct message\n"
                        << "/broadcast MESSAGE    Send to everyone\n"
                        << "/stats                Selected client stats\n"
                        << "/fast                 Toggle fast mode\n"
                        << "/sendimg FILE         Send image\n"
                        << "/sendimg -f FILE      Send raw image\n"
                        << "/sendvid FILE         Send video\n"
                        << "/sendvid -f FILE      Send video with SHA-256\n"
                        << "/quit                 Disconnect selected client\n"
                        << "/shutdown             Stop server\n";
                }
                else if (command == "/clients") {
                    show_clients();
                }
                else if (command == "/use") {
                    int id;
                    if (!(input >> id)) {
                        Neon::error("Usage: /use ID");
                        continue;
                    }

                    if (find_client(id)) {
                        selected_id = id;
                        Neon::info("Selected client " + std::to_string(id));
                    } else {
                        Neon::error("Client not found");
                    }
                }
                else if (command == "/to") {
                    int id;
                    if (!(input >> id)) {
                        Neon::error("Usage: /to ID MESSAGE");
                        continue;
                    }

                    std::string message;
                    std::getline(input >> std::ws, message);

                    auto c = find_client(id);
                    if (!c || message.empty()) {
                        Neon::error("Client not found or message empty");
                        continue;
                    }

                    send_text(c, libraries, message);
                }
                else if (command == "/broadcast") {
                    std::string message;
                    std::getline(input >> std::ws, message);

                    if (message.empty())
                        continue;

                    std::vector<std::shared_ptr<Client>> snapshot;
                    {
                        std::lock_guard<std::mutex> lock(clients_mutex);
                        snapshot = clients;
                    }

                    for (auto& c : snapshot) {
                        if (!c->connected)
                            continue;

                        try {
                            send_text(c, libraries, message);
                        } catch (...) {
                            disconnect_client(c);
                        }
                    }
                }
                else if (command == "/stats") {
                    auto c = find_client(selected_id);
                    if (c)
                        Neon::show_stats(c->stats);
                    else
                        Neon::error("No active client selected");
                }
                else if (command == "/fast") {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("Select a client first");
                        continue;
                    }

                    c->fast = !c->fast;
                    Neon::info(std::string("Fast mode: ") +
                               (c->fast ? "ON" : "OFF"));
                }
                else if (command == "/sendimg" ||
                         command == "/sendvid") {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("Select a client first");
                        continue;
                    }

                    std::string option, path;
                    input >> option;

                    bool raw = option == "-f";
                    if (raw)
                        input >> path;
                    else
                        path = option;

                    if (path.empty()) {
                        Neon::error("File path required");
                        continue;
                    }

                    if (command == "/sendimg")
                        transmit_image(c, path, raw);
                    else
                        transmit_video(c, libraries, path, raw);
                }
                else if (command == "/quit") {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("No active client selected");
                        continue;
                    }

                    try {
                        send_text(c, libraries, "/quit");
                    } catch (...) {}

                    disconnect_client(c);
                    selected_id = -1;
                }
                else if (command == "/shutdown") {
                    running = false;
                    break;
                }
                else {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("Select a client using /use ID");
                        continue;
                    }

                    send_text(c, libraries, line);
                }
            }
            catch (const std::exception& e) {
                Neon::error(e.what());

                auto c = find_client(selected_id);
                if (c && !c->connected)
                    selected_id = -1;
            }
        }

        running = false;
        shutdown(server, SHUT_RDWR);

        {
            std::lock_guard<std::mutex> lock(clients_mutex);
            for (auto& c : clients)
                disconnect_client(c);
        }

        if (accept_thread.joinable())
            accept_thread.join();

        cleanup_clients();

        {
            std::lock_guard<std::mutex> lock(clients_mutex);
            for (auto& c : clients) {
                disconnect_client(c);
                if (c->receiver.joinable())
                    c->receiver.join();
                close(c->fd);
            }
            clients.clear();
        }

        close(server);
        Neon::info("CipherAPI server stopped");
        return 0;
    }
    catch (const std::exception& e) {
        std::cerr << "FATAL: " << e.what() << '\n';
        return 1;
    }
}
