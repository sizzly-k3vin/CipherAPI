    AuthStore auth("cipher_test_users.db");
    std::string key, id;

    if (!auth.register_user("test_user", key, id)) return 1;

    std::string verified_id;
    if (!auth.authenticate("test_user", key, verified_id)) return 2;
    if (id != verified_id) return 3;
    {
        AuthStore restarted("cipher_test_users.db");
        std::string after_restart;
        if (!restarted.authenticate("test_user", key, after_restart)) return 4;
        if (after_restart != id) return 5;

    std::cout << "Authentication: PASS\\n";
    std::cout << "UserID consistency: PASS\\n";
    std::cout << "Restart persistence: PASS\\n";

from pathlib import Path
import shutil

ROOT = Path(__file__).resolve().parent
SRC = ROOT / "src"
INCLUDE = ROOT / "include"
CMAKE = ROOT / "CMakeLists.txt"

if not SRC.is_dir():
    raise SystemExit("ERROR: src/ directory not found")

# Discover C++ files
sources = sorted(SRC.rglob("*.cpp"))

# Known executable entry points
executables = {
    "client": SRC / "client.cpp",
    "server": SRC / "server.cpp",
}

for name, path in executables.items():
    if not path.is_file():
        raise SystemExit(f"ERROR: Missing executable source: {path}")

# Remaining source files become the shared library
core_sources = [
    p for p in sources
    if p not in executables.values()
]

# Discover directories containing headers
include_dirs = []
if INCLUDE.is_dir():
    include_dirs = sorted({
        p.parent for p in INCLUDE.rglob("*")
        if p.is_file() and p.suffix in {".h", ".hpp"}
    } | {INCLUDE})

def cmake_path(path):
    return "${CMAKE_CURRENT_SOURCE_DIR}/" + path.relative_to(ROOT).as_posix()

lines = [
    "cmake_minimum_required(VERSION 3.20)",
    "",
    "project(CipherAPI VERSION 0.4 LANGUAGES CXX)",
    "",
    "set(CMAKE_CXX_STANDARD 20)",
    "set(CMAKE_CXX_STANDARD_REQUIRED ON)",
    "set(CMAKE_CXX_EXTENSIONS OFF)",
    "",
]

if include_dirs:
    lines.append("set(CIPHERAPI_INCLUDE_DIRS")
    for directory in include_dirs:
        lines.append(f'    "{cmake_path(directory)}"')
    lines.extend([")", ""])

if core_sources:
    lines.append("add_library(cipher_core STATIC")
    for source in core_sources:
        lines.append(f'    "{cmake_path(source)}"')
    lines.extend([
        ")",
        "target_include_directories(cipher_core PUBLIC ${CIPHERAPI_INCLUDE_DIRS})",
        "",
    ])

for name, source in executables.items():
    target = f"cipher_{name}"
    lines.append(f'add_executable({target} "{cmake_path(source)}")')

    if include_dirs:
        lines.append(
            f"target_include_directories({target} PRIVATE ${'{'}CIPHERAPI_INCLUDE_DIRS{'}'})"
        )

    if core_sources:
        lines.append(f"target_link_libraries({target} PRIVATE cipher_core)")

    lines.append("")

# Back up existing CMake configuration
if CMAKE.exists():
    shutil.copy2(CMAKE, ROOT / "CMakeLists.txt.bak")

CMAKE.write_text("\n".join(lines) + "\n")

print("CipherAPI CMake configuration generated!")
print(f"Discovered C++ files: {len(sources)}")
print(f"Core library sources: {len(core_sources)}")
print(f"Header directories: {len(include_dirs)}")
print("Executables: cipher_client, cipher_server")
print("Generated: CMakeLists.txt")

namespace CipherDebug {
inline std::mutex mutex;

inline bool enabled() {
    const char* v = std::getenv("CIPHER_DEBUG");
    return v && std::string_view(v) == "1";
}

inline void log(std::string_view level, std::string_view msg,
                const std::source_location loc =
                    std::source_location::current()) {

    auto now = std::chrono::system_clock::now();
    auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(
        now.time_since_epoch()) % 1000;

    std::lock_guard lock(mutex);
    std::cerr << "[" << level << "] "
              << std::chrono::system_clock::to_time_t(now)
              << "." << ms.count()
              << " [T:" << std::this_thread::get_id() << "] "
              << loc.file_name() << ":" << loc.line()
              << " | " << loc.function_name()
              << " | " << msg << '\n';
}
}

namespace CipherLog {
inline std::mutex log_mutex;

inline void write(const char* level, const std::string& message,
                  std::ostream& output) {
    const auto now = std::chrono::system_clock::now();
    const auto time = std::chrono::system_clock::to_time_t(now);
    std::tm local{};
    localtime_r(&time, &local);

    std::lock_guard<std::mutex> lock(log_mutex);
    output << '[' << std::put_time(&local, "%H:%M:%S")
           << "] [" << level << "] " << message << '\n';
    output.flush();
}


inline const char* errno_name(int code) {
    if (code == EADDRINUSE) return "EADDRINUSE";
    if (code == EADDRNOTAVAIL) return "EADDRNOTAVAIL";
    if (code == EACCES) return "EACCES";
    if (code == EBADF) return "EBADF";
    if (code == EINTR) return "EINTR";
    if (code == EINVAL) return "EINVAL";
    if (code == EMFILE) return "EMFILE";
    if (code == ENFILE) return "ENFILE";
    if (code == ENOMEM) return "ENOMEM";
    if (code == ENOBUFS) return "ENOBUFS";
    if (code == ECONNREFUSED) return "ECONNREFUSED";
    if (code == ECONNRESET) return "ECONNRESET";
    if (code == ETIMEDOUT) return "ETIMEDOUT";
    if (code == EPIPE) return "EPIPE";
    if (code == ENOTCONN) return "ENOTCONN";
    return "UNKNOWN";
}

inline void system_error(const std::string& id,
                         const std::string& operation,
                         int code) {
    const std::string detail =
        std::error_code(code, std::generic_category()).message();

    write("ERROR",
          "[" + id + "] " + operation +
          " | errno=" + std::to_string(code) +
          " | name=" + errno_name(code) +
          " | detail=" + detail,
          std::cerr);
}

#pragma once

#include <arpa/inet.h>
#include <sys/socket.h>
#include <unistd.h>
#include <cerrno>
#include <cstdint>
#include <stdexcept>
#include <string>

namespace CipherTransport {

enum FrameType : uint8_t {
    TEXT = 1,
    IMAGE = 2,
    FAST_TEXT = 3,
    RAW_IMAGE = 4,
    VIDEO_START = 5,
    VIDEO_DATA = 6,
    VIDEO_END = 7,
    AUTH_REGISTER = 8,
    AUTH_LOGIN = 9,
    AUTH_OK = 10,
    AUTH_FAIL = 11
};

constexpr uint32_t MAX_TEXT_FRAME = 1024 * 1024;
constexpr uint32_t MAX_IMAGE_FRAME = 32 * 1024 * 1024;
constexpr uint32_t MAX_VIDEO_FRAME = 4 * 1024 * 1024;

inline bool valid_type(uint8_t type) {
    return type == TEXT || type == IMAGE ||
           type == FAST_TEXT || type == RAW_IMAGE ||
           type == VIDEO_START || type == VIDEO_DATA ||
           type == VIDEO_END || type == AUTH_REGISTER ||
           type == AUTH_LOGIN || type == AUTH_OK ||
           type == AUTH_FAIL;
}

inline uint32_t frame_limit(uint8_t type) {
    if (type == VIDEO_START || type == VIDEO_DATA ||
        type == VIDEO_END)
        return MAX_VIDEO_FRAME;

    return (type == IMAGE || type == RAW_IMAGE)
        ? MAX_IMAGE_FRAME : MAX_TEXT_FRAME;
}

inline bool receive_exact(int fd, void* data, size_t size) {
    char* ptr = static_cast<char*>(data);
    size_t received = 0;

    while (received < size) {
        ssize_t n = recv(fd, ptr + received, size - received, 0);
        if (n < 0 && errno == EINTR) continue;
        if (n == 0 && received == 0) return false;
        if (n <= 0) throw std::runtime_error("Incomplete TCP frame");
        received += static_cast<size_t>(n);
    }
    return true;
}

inline void send_exact(int fd, const void* data, size_t size) {
    const char* ptr = static_cast<const char*>(data);
    size_t sent = 0;

    while (sent < size) {
#ifdef MSG_NOSIGNAL
        ssize_t n = ::send(fd, ptr + sent, size - sent, MSG_NOSIGNAL);
#else
        ssize_t n = ::send(fd, ptr + sent, size - sent, 0);
#endif
        if (n < 0 && errno == EINTR) continue;
        if (n <= 0) throw std::runtime_error("TCP send failed");
        sent += static_cast<size_t>(n);
    }
}

inline void send_frame(int fd, FrameType type,
                       const std::string& payload) {
    uint8_t packet_type = static_cast<uint8_t>(type);

    if (!valid_type(packet_type))
        throw std::runtime_error("Unknown outgoing frame type");

    if (payload.size() > frame_limit(packet_type))
        throw std::runtime_error("Frame exceeds size limit");

    if (type == TEXT && payload.size() % 64 != 0)
        throw std::runtime_error("Invalid encoded text payload");

    uint32_t length = htonl(static_cast<uint32_t>(payload.size()));

    send_exact(fd, &packet_type, sizeof(packet_type));
    send_exact(fd, &length, sizeof(length));

    if (!payload.empty())
        send_exact(fd, payload.data(), payload.size());
}

inline bool receive_frame(int fd, uint8_t& type,
                          std::string& payload) {
    uint32_t network_length = 0;

    if (!receive_exact(fd, &type, sizeof(type)))
        return false;

    if (!receive_exact(fd, &network_length, sizeof(network_length)))
        throw std::runtime_error("Incomplete frame header");

    if (!valid_type(type))
        throw std::runtime_error("Unknown packet type");

    uint32_t length = ntohl(network_length);

    if (length > frame_limit(type))
        throw std::runtime_error("Incoming frame exceeds size limit");

    if (type == TEXT && length % 64 != 0)
        throw std::runtime_error("Invalid encoded text frame");

    payload.resize(length);

    if (length && !receive_exact(fd, payload.data(), length))
        throw std::runtime_error("Incomplete frame payload");

    return true;
}

}

    auto result = CipherVideo::send_file(
        fd, libraries, path, use_sha256
    );

    const auto end = std::chrono::steady_clock::now();
    const double seconds =
        std::chrono::duration<double>(end - start).count();

    ++stats.sent_messages;
    stats.sent_bytes += result.wire_bytes;

    Neon::info(
        "CVS/1 | " + std::to_string(result.original_bytes) +
        " original bytes | " + std::to_string(result.wire_bytes) +
        " wire bytes | " + std::to_string(result.chunks) +
        " chunks | " + std::to_string(seconds) + " seconds"
    );

    Neon::send(
        use_sha256
            ? "Video transmitted with SHA-256"
            : "Video transmitted without SHA-256"
    );
}

'''

def replace_once(source, pattern, replacement, label):
    updated, count = re.subn(pattern, replacement, source, count=1,
                             flags=re.MULTILINE)
    if count != 1:
        raise RuntimeError(f"{label}: expected one match, found {count}")
    return updated

def patch_file(path, main_pattern, command_fd):
    source = path.read_text()

    if '#include "video.hpp"' in source:
        raise RuntimeError(f"{path}: video.hpp already included; refusing duplicate patch")

    source = replace_once(
        source,
        r'^#include "neon\.hpp"\s*$',
        '#include "neon.hpp"\n#include "video.hpp"',
        f"{path}: include"
    )

    source = replace_once(
        source,
        main_pattern,
        sender + r'\g<0>',
        f"{path}: sender function"
    )

    # Add a persistent receiver object inside the receiving thread.
    thread_pattern = r'^([ \t]*)std::thread receiver\(\[&\]\(\) \{\s*try \{'
    def thread_replacement(match):
        indent = match.group(1)
        return (
            indent + 'std::thread receiver([&]() {\n'
            + indent + '    CipherVideo::Receiver video_receiver(libraries);\n'
            + indent + '    try {'
        )

    source = replace_once(
        source, thread_pattern, thread_replacement,
        f"{path}: receiver object"
    )

    # Add the CVS/1 receive branch before the existing IMAGE branch.
    image_pattern = r'^([ \t]*)else if \(type == CipherTransport::IMAGE\) \{'
    def image_replacement(match):
        indent = match.group(1)
        branch = (
            indent + 'else if (type == CipherTransport::VIDEO_START ||\n'
            + indent + '         type == CipherTransport::VIDEO_DATA ||\n'
            + indent + '         type == CipherTransport::VIDEO_END) {\n'
            + indent + '    if (type == CipherTransport::VIDEO_START)\n'
            + indent + '        Neon::receive("CVS/1 video transfer started");\n\n'
            + indent + '    std::string saved = video_receiver.handle(type, payload);\n\n'
            + indent + '    if (!saved.empty()) {\n'
            + indent + '        ++stats.received_messages;\n'
            + indent + '        Neon::decode("CVS/1 video reconstructed");\n'
            + indent + '        Neon::print_message("VIDEO", "Saved to " + saved);\n'
            + indent + '    }\n'
            + indent + '}\n'
        )
        return branch + match.group(0)

    source = replace_once(
        source, image_pattern, image_replacement,
        f"{path}: video receive branch"
    )

    # Add /sendvid before /sendimg.
    command_pattern = r'^([ \t]*)if \(message\.rfind\("/sendimg ", 0\) == 0\) \{'
    def command_replacement(match):
        indent = match.group(1)
        inner = indent + '    '
        block = (
            indent + 'if (message.rfind("/sendvid ", 0) == 0) {\n'
            + inner + 'std::string argument = message.substr(9);\n'
            + inner + 'bool no_sha = argument.rfind("-f ", 0) == 0;\n'
            + inner + 'std::string path = no_sha ? argument.substr(3) : argument;\n\n'
            + inner + 'if (path.size() >= 2 && path.front() == \'"\' && path.back() == \'"\')\n'
            + inner + '    path = path.substr(1, path.size() - 2);\n\n'
            + inner + 'try {\n'
            + inner + '    transmit_video(' + command_fd + ', libraries, stats, path, !no_sha);\n'
            + inner + '} catch (const std::exception& e) {\n'
            + inner + '    Neon::error(e.what());\n'
            + inner + '}\n'
            + inner + 'continue;\n'
            + indent + '}\n\n'
        )
        return block + match.group(0)

    source = replace_once(
        source, command_pattern, command_replacement,
        f"{path}: sendvid command"
    )

    return source

def main():
    if "video.cpp" not in Path("CMakeLists.txt").read_text():
        raise RuntimeError("CMakeLists.txt does not include src/video.cpp")

    transport = Path("include/transport.hpp").read_text()
    for frame in ("VIDEO_START", "VIDEO_DATA", "VIDEO_END"):
        if frame not in transport:
            raise RuntimeError(f"Transport is missing {frame}")

    client_original = CLIENT.read_text()
    server_original = SERVER.read_text()

    # Build both patched versions in memory before writing either.
    client_new = patch_file(
        CLIENT,
        r'^int main\(int argc, char\* argv\[\]\) \{',
        "sock"
    )
    server_new = patch_file(
        SERVER,
        r'^int main\(\) \{',
        "client"
    )

    # Backups, then writes.
    CLIENT.with_suffix(".cpp.pre_video.bak").write_text(client_original)
    SERVER.with_suffix(".cpp.pre_video.bak").write_text(server_original)

    CLIENT.write_text(client_new)
    SERVER.write_text(server_new)

    print("CVS/1 integration patch applied successfully.")
    print("Backups: src/client.cpp.pre_video.bak")
    print("         src/server.cpp.pre_video.bak")

if __name__ == "__main__":
    main()

def _error():
    value = _lib.capi_last_error()
    return value.decode(errors="replace") if value else "Unknown error"


class _SendMode:
    def __init__(self, client, fast):
        self._client = client
        self._fast = fast

    def __call__(self, message):
        return self._client._send(message, fast=self._fast)


class _SendDispatcher:
    def __init__(self, client):
        self._client = client
        self.ft = _SendMode(client, True)
        self.ff = _SendMode(client, False)

    def __call__(self, message, fast=False):
        return self._client._send(message, fast=fast)


class Client:
    def __init__(self, host="127.0.0.1", port=8080, library_dir=None):
        self._handle = None
        self._closed = False
        self.send = _SendDispatcher(self)
        directory = Path(library_dir or ROOT / "libraries")
        self._handle = _lib.capi_client_create(
            host.encode(), port, str(directory).encode()
        )
        if not self._handle:
            raise RuntimeError(_error())

    def connect(self):
        if self._closed:
            raise RuntimeError("Client is closed")
        if _lib.capi_client_connect(self._handle) != 0:
            raise RuntimeError(_error())
        return self

    def _send(self, message, fast=False):
        if self._closed:
            raise RuntimeError("Client is closed")
        if not isinstance(message, str):
            raise TypeError("message must be a string")
        if "\x00" in message:
            raise ValueError("NUL characters are not supported")
        if _lib.capi_client_send_text(
            self._handle, message.encode("utf-8"), int(bool(fast))
        ) != 0:
            raise RuntimeError(_error())

    def send_image(self, path, raw=False):
        if self._closed:
            raise RuntimeError("Client is closed")
        result = _lib.capi_client_send_image(
            self._handle, os.fsencode(path), int(bool(raw))
        )
        if result != 0:
            raise RuntimeError(_error())

    def send_video(self, path, checksum=True):
        if self._closed:
            raise RuntimeError("Client is closed")
        result = _lib.capi_client_send_video(
            self._handle, os.fsencode(path), int(bool(checksum))
        )
        if result != 0:
            raise RuntimeError(_error())

    def receive_event(self, timeout=1.0):
        if self._closed:
            raise RuntimeError("Client is closed")

        timeout_ms = -1 if timeout is None else max(0, int(timeout * 1000))
        buffer = ctypes.create_string_buffer(1024 * 1024 + 1)
        event_type = ctypes.c_int()

        result = _lib.capi_client_receive_event(
            self._handle,
            ctypes.byref(event_type),
            buffer,
            ctypes.sizeof(buffer),
            timeout_ms
        )

        if result == -2:
            return None
        if result < 0:
            raise RuntimeError(_error())

        value = buffer.raw[:result].decode("utf-8")
        names = {1: "text", 2: "image", 3: "video"}
        return names.get(event_type.value, "unknown"), value

    def receive(self, timeout=1.0):
        if self._closed:
            raise RuntimeError("Client is closed")
        timeout_ms = -1 if timeout is None else max(0, int(timeout * 1000))
        buffer = ctypes.create_string_buffer(1024 * 1024 + 1)
        result = _lib.capi_client_receive(
            self._handle, buffer, ctypes.sizeof(buffer), timeout_ms
        )
        if result == -2:
            return None
        if result < 0:
            raise RuntimeError(_error())
        return buffer.raw[:result].decode("utf-8")

    def close(self):
        if not self._closed:
            _lib.capi_client_close(self._handle)
            _lib.capi_client_destroy(self._handle)
            self._handle = None
            self._closed = True

    def __enter__(self):
        return self.connect()

    def __exit__(self, *args):
        self.close()

    def __del__(self):
        try:
            self.close()
        except Exception:
            pass

struct CAPIClient {
    std::string host;
    uint16_t port;
    std::unique_ptr<LibraryManager> libraries;
    std::unique_ptr<CipherVideo::Receiver> video_receiver;
    int fd = -1;
    std::mutex send_mutex;
    std::mutex receive_mutex;

static thread_local std::string last_error;

static void set_error(const char* message) {
    last_error = message ? message : "Unknown CipherAPI error";
}

static int wait_readable(int fd, int timeout_ms) {
    pollfd descriptor{fd, POLLIN, 0};
    int result;

    do {
        result = poll(&descriptor, 1, timeout_ms);
    } while (result < 0 && errno == EINTR);

    if (result < 0)
        throw std::runtime_error("Socket poll failed");

    return result;
}

extern "C" {

const char* capi_last_error(void) {
    return last_error.c_str();
}

CAPIClient* capi_client_create(
    const char* host, uint16_t port, const char* library_directory
) {
    try {
        if (!host || !*host || !library_directory || !*library_directory)
            throw std::invalid_argument("Host and library directory are required");

        last_error.clear();
        return new CAPIClient(host, port, library_directory);
    } catch (const std::exception& e) {
        set_error(e.what());
        return nullptr;
    } catch (...) {
        set_error("Unknown error creating CipherAPI client");
        return nullptr;
    }
}

int capi_client_connect(CAPIClient* client) {
    try {
        if (!client)
            throw std::invalid_argument("Invalid client handle");

        if (client->fd >= 0)
            return 0;

        addrinfo hints{};
        hints.ai_family = AF_UNSPEC;
        hints.ai_socktype = SOCK_STREAM;

        addrinfo* result = nullptr;
        const std::string service = std::to_string(client->port);

        int status = getaddrinfo(
            client->host.c_str(), service.c_str(), &hints, &result
        );

        if (status != 0)
            throw std::runtime_error(gai_strerror(status));

        int connected_fd = -1;

        for (addrinfo* item = result; item; item = item->ai_next) {
            int candidate = socket(
                item->ai_family, item->ai_socktype, item->ai_protocol
            );

            if (candidate < 0)
                continue;

            if (::connect(candidate, item->ai_addr, item->ai_addrlen) == 0) {
                connected_fd = candidate;
                break;
            }

            ::close(candidate);
        }

        freeaddrinfo(result);

        if (connected_fd < 0)
            throw std::runtime_error("Could not connect to CipherAPI server");

        client->fd = connected_fd;
        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown connection error");
        return -1;
    }
}

int capi_client_send_text(CAPIClient* client, const char* message, int fast) {
    try {
        if (!client || !message)
            throw std::invalid_argument("Invalid client or message");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->send_mutex);
        std::string text(message);

        if (fast) {
            CipherTransport::send_frame(
                client->fd, CipherTransport::FAST_TEXT, text
            );
        } else {
            std::string encoded = client->libraries->encode(text);
            CipherTransport::send_frame(
                client->fd, CipherTransport::TEXT, encoded
            );
        }

        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown text send error");
        return -1;
    }
}

int capi_client_send_image(CAPIClient* client, const char* path, int raw) {
    try {
        if (!client || !path || !*path)
            throw std::invalid_argument("Invalid image path");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->send_mutex);

        std::string packet = raw
            ? ImageLibrary::pack_raw(path)
            : ImageLibrary::pack(path);

        auto type = raw
            ? CipherTransport::RAW_IMAGE
            : CipherTransport::IMAGE;

        CipherTransport::send_frame(client->fd, type, packet);

        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown image send error");
        return -1;
    }
}

int capi_client_send_video(CAPIClient* client, const char* path, int checksum) {
    try {
        if (!client || !path || !*path)
            throw std::invalid_argument("Invalid video path");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->send_mutex);

        CipherVideo::send_file(
            client->fd, *client->libraries, path, checksum != 0, false

        last_error.clear();
        return 0;
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown video send error");
        return -1;
    }
}

int capi_client_receive_event(
    CAPIClient* client, int* event_type,
    char* buffer, size_t capacity, int timeout_ms
) {
    try {
        if (!client || !event_type || !buffer || capacity == 0)
            throw std::invalid_argument("Invalid receive arguments");

        if (client->fd < 0)
            throw std::runtime_error("Client is not connected");

        std::lock_guard<std::mutex> lock(client->receive_mutex);

        while (true) {
            int ready = wait_readable(client->fd, timeout_ms);

            if (ready == 0)
                return -2;

            uint8_t type = 0;
            std::string payload;

            if (!CipherTransport::receive_frame(client->fd, type, payload))
                throw std::runtime_error("Server closed the connection");

            int kind = 0;
            std::string output;

            if (type == CipherTransport::TEXT) {
                output = client->libraries->decode(payload);
                kind = 1;
            } else if (type == CipherTransport::FAST_TEXT) {
                output = std::move(payload);
                kind = 1;
            } else if (type == CipherTransport::IMAGE) {
                output = ImageLibrary::unpack(
                    payload, "received_images"
                );
                kind = 2;
            } else if (type == CipherTransport::RAW_IMAGE) {
                output = ImageLibrary::unpack_raw(
                    payload, "received_images"
                );
                kind = 2;
            } else if (
                type >= CipherTransport::VIDEO_START &&
                type <= CipherTransport::VIDEO_END
            ) {
                output = client->video_receiver->handle(type, payload);

                if (output.empty())
                    continue;

                kind = 3;
            } else {
                throw std::runtime_error("Unsupported incoming frame type");
            }

            if (output.size() + 1 > capacity)
                throw std::runtime_error("Receive buffer is too small");

            std::memcpy(buffer, output.data(), output.size());
            buffer[output.size()] = 0;
            *event_type = kind;

            last_error.clear();
            return static_cast<int>(output.size());
        }
    } catch (const std::exception& e) {
        set_error(e.what());
        return -1;
    } catch (...) {
        set_error("Unknown receive error");
        return -1;
    }
}

int capi_client_receive(
    CAPIClient* client, char* buffer, size_t capacity, int timeout_ms
) {
    int event_type = 0;

    int result = capi_client_receive_event(
        client, &event_type, buffer, capacity, timeout_ms
    );

    if (result < 0)
        return result;

    if (event_type != 1) {
        set_error("Received media event; use receive_event() instead");
        return -1;
    }

    return result;
}

void capi_client_close(CAPIClient* client) {
    if (!client || client->fd < 0)
        return;

    shutdown(client->fd, SHUT_RDWR);
    ::close(client->fd);
    client->fd = -1;
}

void capi_client_destroy(CAPIClient* client) {
    if (!client)
        return;

    capi_client_close(client);
    delete client;
}

AuthStore::AuthStore(const std::string& file) : path(file) {
    std::ifstream input(path);
    std::string username, first, second;

    while (input >> username >> first) {
        UserRecord record;

        if (first.size() == 64) {
            record.user_id = generate_user_id();
            record.ckey_hash = first;
        } else {
            record.user_id = first;
            if (!(input >> record.ckey_hash))
                break;
        }

        if (valid_username(username) &&
            record.user_id.size() == 32 &&
            record.ckey_hash.size() == 64) {
            users[username] = record;
        }
    }

    if (std::filesystem::exists(path)) {
        chmod(path.c_str(), 0600);
        save_locked();
    }
}

bool AuthStore::valid_username(const std::string& username) {
    if (username.size() < 3 || username.size() > 24)
        return false;

    for (unsigned char c : username) {
        if (!std::isalnum(c) && c != 95 && c != 45 && c != 46)
            return false;
    }
    return true;
}

std::string AuthStore::hash_key(const std::string& key) {
    unsigned char digest[SHA256_DIGEST_LENGTH];
    SHA256(reinterpret_cast<const unsigned char*>(key.data()),
           key.size(), digest);

    std::ostringstream output;
    output << std::hex << std::setfill(static_cast<char>(48));

    for (unsigned char byte : digest)
        output << std::setw(2) << static_cast<unsigned int>(byte);

    return output.str();
}

std::string AuthStore::generate_user_id() {
    std::array<unsigned char, 16> bytes{};

    if (RAND_bytes(bytes.data(), static_cast<int>(bytes.size())) != 1)
        throw std::runtime_error("Secure UserID generation failed");

    std::ostringstream output;
    output << std::hex << std::setfill(static_cast<char>(48));

    for (unsigned char byte : bytes)
        output << std::setw(2) << static_cast<unsigned int>(byte);

    return output.str();
}

void AuthStore::save_locked() const {
    const std::string temporary = path + ".tmp";
    std::ofstream output(temporary, std::ios::trunc);

    if (!output)
        throw std::runtime_error("Cannot write user database");

    for (const auto& entry : users)
        output << entry.first << " " << entry.second.user_id
               << " " << entry.second.ckey_hash << "\n";

    output.flush();
    if (!output)
        throw std::runtime_error("User database write failed");

    output.close();
    chmod(temporary.c_str(), 0600);

    if (std::rename(temporary.c_str(), path.c_str()) != 0)
        throw std::runtime_error("Cannot replace user database");

    chmod(path.c_str(), 0600);
}

bool AuthStore::register_user(
    const std::string& username, std::string& ckey
) {
    std::string user_id;
    return register_user(username, ckey, user_id);
}

bool AuthStore::register_user(
    const std::string& username, std::string& ckey,
    std::string& user_id
) {
    std::lock_guard<std::mutex> lock(mutex);

    if (!valid_username(username) || users.count(username))
        return false;

    std::array<unsigned char, 32> random_bytes{};

    if (RAND_bytes(random_bytes.data(),
                   static_cast<int>(random_bytes.size())) != 1)
        throw std::runtime_error("Secure CKey generation failed");

    std::ostringstream token;
    token << std::hex << std::setfill(static_cast<char>(48));

    for (unsigned char byte : random_bytes)
        token << std::setw(2) << static_cast<unsigned int>(byte);

    ckey = token.str();
    user_id = generate_user_id();

    users[username] = {user_id, hash_key(ckey)};

    try {
        save_locked();
    } catch (...) {
        users.erase(username);
        ckey.clear();
        user_id.clear();
        throw;
    }

    return true;
}

bool AuthStore::authenticate(
    const std::string& username, const std::string& ckey
) const {
    std::string user_id;
    return authenticate(username, ckey, user_id);
}

bool AuthStore::authenticate(
    const std::string& username, const std::string& ckey,
    std::string& user_id
) const {
    std::lock_guard<std::mutex> lock(mutex);

    auto found = users.find(username);
    if (found == users.end() || ckey.size() != 64)
        return false;

    std::string supplied = hash_key(ckey);

    if (CRYPTO_memcmp(found->second.ckey_hash.data(),
                      supplied.data(), supplied.size()) != 0)
        return false;

    user_id = found->second.user_id;
    return true;
}

static void transmit(int fd, LibraryManager& libraries,
                     Neon::Stats& stats, const std::string& message) {
    const auto start = std::chrono::steady_clock::now();
    std::string encoded = libraries.encode(message);
    const auto end = std::chrono::steady_clock::now();

    const double ms =
        std::chrono::duration<double, std::milli>(end - start).count();

    Neon::encode(
        std::to_string(message.size()) + " chars | " +
        std::to_string(encoded.size() / 64) + " tokens | " +
        std::to_string(encoded.size()) + " bytes | " +
        std::to_string(ms) + " ms"
    );

    CipherTransport::send_frame(fd, CipherTransport::TEXT, encoded);

    ++stats.sent_messages;
    stats.sent_chars += message.size();
    stats.sent_tokens += encoded.size() / 64;
    stats.sent_bytes += encoded.size();

    Neon::send("Text frame transmitted");
}

static void transmit_fast(int fd, Neon::Stats& stats,
                          const std::string& message) {
    CipherTransport::send_frame(fd, CipherTransport::FAST_TEXT, message);

    ++stats.sent_messages;
    stats.sent_chars += message.size();
    stats.sent_bytes += message.size();

    Neon::info("FAST text | " + std::to_string(message.size()) + " raw bytes");
    Neon::send("Binary text frame transmitted");
}

static void transmit_image(int fd, Neon::Stats& stats,
                           const std::string& path) {
    const auto start = std::chrono::steady_clock::now();

    std::string packet = ImageLibrary::pack(path);

    const auto packed = std::chrono::steady_clock::now();

    CipherTransport::send_frame(fd, CipherTransport::IMAGE, packet);

    const auto sent = std::chrono::steady_clock::now();

    const double pack_ms =
        std::chrono::duration<double, std::milli>(packed - start).count();

    const double send_ms =
        std::chrono::duration<double, std::milli>(sent - packed).count();

    ++stats.sent_messages;
    stats.sent_bytes += packet.size();

    Neon::info(
        "Image packet | " + std::to_string(packet.size()) +
        " bytes | pack " + std::to_string(pack_ms) +
        " ms | send " + std::to_string(send_ms) + " ms"
    );

    Neon::send("Image transmitted successfully");
}

static void transmit_raw_image(int fd, Neon::Stats& stats,

    double send_ms =

static void transmit_video(int fd, const LibraryManager& libraries,
                           Neon::Stats& stats, const std::string& path,
                           bool use_sha256, bool fast_mode) {
    const auto start = std::chrono::steady_clock::now();

    auto result = CipherVideo::send_file(
        fd, libraries, path, use_sha256, fast_mode
    );

    const auto end = std::chrono::steady_clock::now();
    const double seconds =
        std::chrono::duration<double>(end - start).count();

    ++stats.sent_messages;
    stats.sent_bytes += result.wire_bytes;

    Neon::info(
        "CVS/1 | " + std::to_string(result.original_bytes) +
        " original bytes | " + std::to_string(result.wire_bytes) +
        " wire bytes | " + std::to_string(result.chunks) +
        " chunks | " + std::to_string(seconds) + " seconds"
    );

    Neon::send(
        use_sha256
            ? "Video transmitted with SHA-256"
            : "Video transmitted without SHA-256"
    );
}

int main(int argc, char* argv[]) {
    int sock = -1;

    try {
        Neon::banner("CLIENT");

        LibraryManager libraries("libraries");
        Neon::info("Translation engine initialized");
        Neon::info("Binary image transport initialized");

        sock = socket(AF_INET, SOCK_STREAM, 0);
        if (sock < 0)
            throw std::runtime_error("Socket creation failed");

        sockaddr_in server{};
        server.sin_family = AF_INET;
        server.sin_port = htons(PORT);
        const char* server_ip = argc > 1 ? argv[1] : "127.0.0.1";
        if (inet_pton(AF_INET, server_ip, &server.sin_addr) != 1)
            throw std::runtime_error("Invalid IPv4 address");

        Neon::network(std::string("Connecting to ") +
                     server_ip + ":8080");

        if (connect(sock, reinterpret_cast<sockaddr*>(&server),
                    sizeof(server)) < 0)
            throw std::runtime_error("Connection failed");

        Neon::network("Connection established");
        Neon::info("Full-duplex text + binary communication enabled");
        Neon::show_help_hint();

        Neon::Stats stats;
            bool fast_mode = false;
        std::atomic<bool> connected{true};

        std::thread receiver([&]() {
            CipherVideo::Receiver video_receiver(libraries);
            try {
                uint8_t type = 0;
                std::string payload;

                while (connected &&
                       CipherTransport::receive_frame(sock, type, payload)) {
                    stats.received_bytes += payload.size();

                    if (type == CipherTransport::TEXT) {
                        stats.received_tokens += payload.size() / 64;

                        Neon::receive(
                            std::to_string(payload.size()) + " bytes | " +
                            std::to_string(payload.size() / 64) + " tokens"
                        );

                        const auto start = std::chrono::steady_clock::now();
                        std::string message = libraries.decode(payload);
                        const auto end = std::chrono::steady_clock::now();

                        const double ms =
                            std::chrono::duration<double, std::milli>(
                                end - start).count();

                        ++stats.received_messages;
                        stats.received_chars += message.size();

                        Neon::decode(
                            std::to_string(payload.size() / 64) +
                            " tokens -> " + std::to_string(message.size()) +
                            " chars | " + std::to_string(ms) + " ms"
                        );

                        Neon::print_message("SERVER", message);

                        if (message == "/quit") {
                            connected = false;
                            shutdown(sock, SHUT_RDWR);
                            break;
                        }
                    }
                    else if (type == CipherTransport::FAST_TEXT) {
                            ++stats.received_messages;
                            stats.received_chars += payload.size();
                            Neon::receive("FAST text | " +
                                          std::to_string(payload.size()) +
                                          " raw bytes");
                            Neon::print_message("PEER", payload);

                            if (payload == "/quit") {
                                connected = false;
                                shutdown(sock, SHUT_RDWR);
                                break;
                            }
                        }
                        else if (type == CipherTransport::RAW_IMAGE) {
                            Neon::receive("Raw image | " +
                                          std::to_string(payload.size()) +
                                          " bytes | no checksum");

                            std::string saved = ImageLibrary::unpack_raw(payload);
                            ++stats.received_messages;

                            Neon::info("Raw image saved without SHA-256 verification");
                            Neon::print_message("IMAGE", "Saved to " + saved);
                        }
                        else if (type == CipherTransport::VIDEO_START ||
                                 type == CipherTransport::VIDEO_DATA ||
                                 type == CipherTransport::VIDEO_END) {
                            if (type == CipherTransport::VIDEO_START)
                                Neon::receive("CVS/1 video transfer started");

                            std::string saved = video_receiver.handle(type, payload);

                            if (!saved.empty()) {
                                ++stats.received_messages;
                                Neon::decode("CVS/1 video reconstructed");
                                Neon::print_message("VIDEO", "Saved to " + saved);
                            }
                        }
                        else if (type == CipherTransport::IMAGE) {
                        Neon::receive(
                            "Binary image packet | " +
                            std::to_string(payload.size()) + " bytes"
                        );

                        const auto start = std::chrono::steady_clock::now();
                        std::string saved =
                            ImageLibrary::unpack(payload);
                        const auto end = std::chrono::steady_clock::now();

                        const double ms =
                            std::chrono::duration<double, std::milli>(
                                end - start).count();

                        ++stats.received_messages;

                        Neon::decode(
                            "SHA-256 verified | " +
                            std::to_string(payload.size()) +
                            " bytes | " + std::to_string(ms) + " ms"
                        );

                        Neon::print_message("IMAGE", "Saved to " + saved);
                    }
                }
            } catch (const std::exception& e) {
                if (connected) Neon::error(e.what());
            }

            connected = false;
            shutdown(sock, SHUT_RDWR);
            Neon::network("Receive channel closed");
        });

        std::string message;

        while (connected) {
            Neon::prompt("CLIENT");

            if (!std::getline(std::cin, message)) {
                if (connected) {
                    try { transmit(sock, libraries, stats, "/quit"); }
                    catch (...) {}
                }
                connected = false;
                shutdown(sock, SHUT_RDWR);
                break;
            }

            if (message == "/help") {
                Neon::help();
                continue;
            }

            if (message == "/stats") {
                Neon::show_stats(stats);
                continue;
            }

            if (message == "/clear") {
                Neon::clear();
                continue;
            }

            if (message == "/fast") {
                fast_mode = !fast_mode;
                Neon::info(std::string("FAST binary text mode ") +
                           (fast_mode ? "ENABLED" : "DISABLED"));
                continue;
            }

            if (message.rfind("/sendvid ", 0) == 0) {
                std::string argument = message.substr(9);
                bool no_sha = argument.rfind("-f ", 0) == 0;
                std::string path = no_sha ? argument.substr(3) : argument;

                if (path.size() >= 2 && path.front() == '"' && path.back() == '"')
                    path = path.substr(1, path.size() - 2);

                try {
                    transmit_video(sock, libraries, stats, path, !no_sha, fast_mode);
                } catch (const std::exception& e) {
                    Neon::error(e.what());
                }
                continue;
            }

            if (message.rfind("/sendimg ", 0) == 0) {
                std::string argument = message.substr(9);
                bool raw_mode = argument.rfind("-f ", 0) == 0;
                std::string path = raw_mode ? argument.substr(3) : argument;

                if (path.size() >= 2 &&
                    path.front() == '"' && path.back() == '"') {
                    path = path.substr(1, path.size() - 2);
                }

                try {
                    if (raw_mode)
                        transmit_raw_image(sock, stats, path);
                    else
                        transmit_image(sock, stats, path);
                } catch (const std::exception& e) {
                    Neon::error(e.what());
                }
                continue;
            }

            if (!connected) break;

            try {
                if (fast_mode)
                    transmit_fast(sock, stats, message);
                else
                    transmit(sock, libraries, stats, message);
            } catch (const std::exception& e) {
                Neon::error(e.what());
                connected = false;
                shutdown(sock, SHUT_RDWR);
                break;
            }

            if (message == "/quit") {
                connected = false;
                shutdown(sock, SHUT_RDWR);
                break;
            }
        }

        if (receiver.joinable()) receiver.join();

        close(sock);
        sock = -1;

        Neon::network("Client session closed");
        Neon::show_stats(stats);

    } catch (const std::exception& e) {
        if (sock >= 0) {
            shutdown(sock, SHUT_RDWR);
            close(sock);
        }

        Neon::error(e.what());
        return 1;
    }

    return 0;
}

constexpr std::size_t CHECKSUM_SIZE = 64;
constexpr std::size_t MAX_IMAGE_SIZE = 32 * 1024 * 1024;

std::string sha256(const std::string& data) {
    std::array<unsigned char, EVP_MAX_MD_SIZE> digest{};
    unsigned int digest_length = 0;

    if (EVP_Digest(
            data.data(), data.size(),
            digest.data(), &digest_length,
            EVP_sha256(), nullptr) != 1) {
        throw std::runtime_error("SHA-256 calculation failed");
    }

    std::ostringstream output;
    output << std::hex << std::setfill('0');

    for (unsigned int i = 0; i < digest_length; ++i)
        output << std::setw(2) << static_cast<unsigned>(digest[i]);

    return output.str();
}

bool valid_image_extension(const std::string& name) {
    std::string ext = fs::path(name).extension().string();

    std::transform(ext.begin(), ext.end(), ext.begin(),
        [](unsigned char c) { return static_cast<char>(std::tolower(c)); });

    return ext == ".png" || ext == ".jpg" ||
           ext == ".jpeg" || ext == ".webp" ||
           ext == ".gif" || ext == ".bmp";
}

uint16_t read_u16(const std::string& data, std::size_t offset) {
    return static_cast<uint16_t>(
        (static_cast<unsigned char>(data[offset]) << 8) |
         static_cast<unsigned char>(data[offset + 1])
    );
}

namespace ImageLibrary {

std::string pack(const std::string& path) {
    fs::path source(path);

    if (!fs::exists(source) || !fs::is_regular_file(source))
        throw std::runtime_error("Image file not found");

    std::string filename = source.filename().string();

    if (filename.empty() || filename.size() > 255)
        throw std::runtime_error("Invalid image filename");

    if (!valid_image_extension(filename))
        throw std::runtime_error("Unsupported image extension");

    const auto file_size = fs::file_size(source);

    if (file_size > MAX_IMAGE_SIZE - 512)
        throw std::runtime_error("Image exceeds 32 MiB limit");

    std::ifstream file(source, std::ios::binary);

    if (!file)
        throw std::runtime_error("Could not open image");

    std::string bytes{
        std::istreambuf_iterator<char>(file),
        std::istreambuf_iterator<char>()
    };

    if (!file.eof() && file.fail())
        throw std::runtime_error("Image read failed");

    if (bytes.empty())
        throw std::runtime_error("Image is empty");

    const std::string checksum = sha256(bytes);

    // Packet: 2-byte filename length | filename | 64-byte checksum | image
    std::string packet;

    uint16_t name_length = static_cast<uint16_t>(filename.size());

    packet.push_back(static_cast<char>((name_length >> 8) & 0xFF));
    packet.push_back(static_cast<char>(name_length & 0xFF));

    packet += filename;
    packet += checksum;
    packet += bytes;

    return packet;
}

std::string unpack(const std::string& packet,
                   const std::string& output_directory) {
    if (packet.size() < 2 + CHECKSUM_SIZE)
        throw std::runtime_error("Image packet is too small");

    const uint16_t name_length = read_u16(packet, 0);

    if (name_length == 0 || name_length > 255)
        throw std::runtime_error("Invalid filename metadata");

    const std::size_t checksum_offset = 2 + name_length;
    const std::size_t image_offset = checksum_offset + CHECKSUM_SIZE;

    if (packet.size() <= image_offset)
        throw std::runtime_error("Image data is missing");

    std::string filename = packet.substr(2, name_length);

    // Strip any directory components supplied by the sender.
    std::replace(filename.begin(), filename.end(), '\\', '/');
    filename = fs::path(filename).filename().string();

    if (filename.empty() || !valid_image_extension(filename))
        throw std::runtime_error("Invalid received image filename");

    const std::string expected =
        packet.substr(checksum_offset, CHECKSUM_SIZE);

    const std::string bytes = packet.substr(image_offset);
    const std::string actual = sha256(bytes);

    if (actual != expected)
        throw std::runtime_error("SHA-256 verification FAILED");

    fs::create_directories(output_directory);

    fs::path destination = fs::path(output_directory) /
                           ("received_" + filename);

    // Avoid overwriting an earlier received image.
    for (unsigned int i = 1; fs::exists(destination); ++i) {
        destination = fs::path(output_directory) /
            ("received_" + std::to_string(i) + "_" + filename);
    }

    std::ofstream output(destination, std::ios::binary);

    if (!output)
        throw std::runtime_error("Could not create received image");

    output.write(bytes.data(), static_cast<std::streamsize>(bytes.size()));

    if (!output)
        throw std::runtime_error("Image write failed");

    return destination.string();
}



std::string pack_raw(const std::string& path) {
    fs::path source(path);

    if (!fs::exists(source) || !fs::is_regular_file(source))
        throw std::runtime_error("Image file not found");

    std::string filename = source.filename().string();

    if (filename.empty() || filename.size() > 255 ||
        !valid_image_extension(filename))
        throw std::runtime_error("Invalid image filename");

    const auto size = fs::file_size(source);
    if (size == 0 || size > MAX_IMAGE_SIZE - 257)
        throw std::runtime_error("Raw image exceeds size limit or is empty");

    std::ifstream file(source, std::ios::binary);
    if (!file) throw std::runtime_error("Could not open image");

    std::string bytes{
        std::istreambuf_iterator<char>(file),
        std::istreambuf_iterator<char>()
    };

    if (file.bad() || bytes.empty())
        throw std::runtime_error("Image read failed");

    std::string packet;
    uint16_t n = static_cast<uint16_t>(filename.size());

    packet.push_back(static_cast<char>((n >> 8) & 0xff));
    packet.push_back(static_cast<char>(n & 0xff));
    packet += filename;
    packet += bytes;

    return packet;
}

std::string unpack_raw(const std::string& packet,
                       const std::string& output_directory) {
    if (packet.size() < 3)
        throw std::runtime_error("Raw image packet is too small");

    uint16_t name_length = read_u16(packet, 0);

    if (name_length == 0 || name_length > 255 ||
        packet.size() <= 2 + name_length)
        throw std::runtime_error("Invalid raw image metadata");

    std::string filename = packet.substr(2, name_length);

    std::replace(filename.begin(), filename.end(), '\\', '/');
    filename = fs::path(filename).filename().string();

    if (filename.empty() || !valid_image_extension(filename))
        throw std::runtime_error("Invalid received image filename");

    const std::size_t offset = 2 + name_length;
    const std::size_t image_size = packet.size() - offset;

    if (image_size > MAX_IMAGE_SIZE)
        throw std::runtime_error("Raw image exceeds size limit");

    fs::create_directories(output_directory);

    fs::path destination =
        fs::path(output_directory) / ("received_" + filename);

    for (unsigned int i = 1; fs::exists(destination); ++i) {
        destination = fs::path(output_directory) /
            ("received_" + std::to_string(i) + "_" + filename);
    }

    std::ofstream output(destination, std::ios::binary);
    if (!output)
        throw std::runtime_error("Could not create received image");

    output.write(packet.data() + offset,
                 static_cast<std::streamsize>(image_size));

    if (!output)
        throw std::runtime_error("Raw image write failed");

    return destination.string();
}

#include "debug.hpp"
#include "library.hpp"
#include "logger.hpp"

#include <cerrno>
#include <filesystem>
#include <fstream>
#include <stdexcept>
#include <string>

namespace fs = std::filesystem;

LibraryManager::LibraryManager(const std::string& directory) {
    load(directory);
}

void LibraryManager::load(const std::string& directory) {
    for (std::size_t i = 0; i < COUNT; ++i) {
        const std::string name =
            "lib" + std::string(i + 1 < 10 ? "0" : "") +
            std::to_string(i + 1) + ".txt";

        const fs::path path = fs::path(directory) / name;

        auto fail = [&](const std::string& code,
                        const std::string& reason) {
            const std::string message =
                "[" + code + "] " + name + ": " + reason;

            CipherLog::error(message);
            throw std::runtime_error(message);
        };

        // Check whether the library exists.
        std::error_code fs_error;
        bool exists = fs::exists(path, fs_error);

        if (fs_error)
            fail("LIB-003", "Filesystem error: " + fs_error.message());

        if (!exists)
            fail("LIB-001", "File not found: " + path.string());

        // Open library.
        errno = 0;
        std::ifstream file(path);

        if (!file) {
            const int err = errno;

            if (err == EACCES)
                fail("LIB-002", "Permission denied");

            if (err == ENOENT)
                fail("LIB-001", "File disappeared before opening");

            fail("LIB-003", "Failed to open file (errno=" +
                            std::to_string(err) + ")");
        }

        std::string line;
        std::size_t entries = 0;
        std::size_t line_number = 0;

        while (std::getline(file, line)) {
            ++line_number;

            // Expected format: X = 64-character hexadecimal token
            auto separator = line.find(" = ");

            if (separator != 1 || line.size() != 68) {
                fail("LIB-004",
                     "Invalid entry format at line " +
                     std::to_string(line_number));
            }

            auto character = static_cast<unsigned char>(line[0]);

            if (character < 32 || character > 126) {
                fail("LIB-004",
                     "Character outside printable ASCII at line " +
                     std::to_string(line_number));
            }

            std::string token = line.substr(separator + 3);

            for (char c : token) {
                if (!((c >= '0' && c <= '9') ||
                      (c >= 'a' && c <= 'f'))) {
                    fail("LIB-004",
                         "Invalid hexadecimal token at line " +
                         std::to_string(line_number));
                }
            }

            if (forward_[i].count(character)) {
                fail("LIB-005",
                     "Duplicate character at line " +
                     std::to_string(line_number));
            }

            if (reverse_.count(token)) {
                fail("LIB-008",
                     "Duplicate token at line " +
                     std::to_string(line_number));
            }

            forward_[i].emplace(character, token);
            reverse_.emplace(token, TokenEntry{character, i});

            ++entries;
        }

        // Detect file read errors.
        if (file.bad() || (!file.eof() && file.fail())) {
            fail("LIB-007", "I/O error while reading file");
        }

        // Every library must contain exactly 95 entries.
        if (entries != 95) {
            fail("LIB-006",
                 "Expected 95 entries, found " +
                 std::to_string(entries));
        }

        CipherLog::info(
            "[LIB] " + name + " loaded: " +
            std::to_string(entries) + " entries"
        );
    }

    CipherLog::info("[LIB] All 10 libraries loaded successfully.");
}

std::string LibraryManager::encode(
    const std::string& message) const {

    std::array<std::size_t, 256> counters{};
    std::string output;
    output.reserve(message.size() * 64);

    for (unsigned char ch : message) {
        if (ch < 32 || ch > 126)
            throw std::invalid_argument("Printable ASCII only for now");

        std::size_t lib = counters[ch] % COUNT;
        output += forward_[lib].at(ch);
        ++counters[ch];
    }

    CipherLog::info(
        "Encoded " + std::to_string(message.size()) +
        " characters into " +
        std::to_string(output.size()) + " token characters."
    );

    return output;
}

std::string LibraryManager::decode(
    const std::string& stream) const {

    if (stream.size() % 64 != 0)
        throw std::invalid_argument("Invalid stream length");

    std::array<std::size_t, 256> counters{};
    std::string output;
    output.reserve(stream.size() / 64);

    for (std::size_t pos = 0; pos < stream.size(); pos += 64) {
        std::string token = stream.substr(pos, 64);
        auto found = reverse_.find(token);

        if (found == reverse_.end())
            throw std::runtime_error("Unknown token");

        const auto& entry = found->second;
        std::size_t expected = counters[entry.character] % COUNT;

        if (entry.library != expected)
            throw std::runtime_error("Library rotation mismatch");

        output += static_cast<char>(entry.character);
        ++counters[entry.character];
    }

    CipherLog::info(
        "Decoded " + std::to_string(output.size()) +
        " characters successfully."
    );

    return output;
}

static void send_text(
    const std::shared_ptr<Client>& c,
    const LibraryManager& libraries,
    const std::string& message
) {
    std::lock_guard<std::mutex> lock(c->send_mutex);

    if (!c->connected)
        throw std::runtime_error("Client disconnected");

    std::string payload = c->fast
        ? message
        : libraries.encode(message);

    auto type = c->fast
        ? CipherTransport::FAST_TEXT
        : CipherTransport::TEXT;

    CipherTransport::send_frame(c->fd, type, payload);

    ++c->stats.sent_messages;
    c->stats.sent_chars += message.size();
    c->stats.sent_bytes += payload.size();

    Neon::print_message("SERVER -> CLIENT " +
                        std::to_string(c->id), message);
}

static void receive_client(
    std::shared_ptr<Client> c,
    const LibraryManager& libraries
) {
    CipherVideo::Receiver video_receiver(libraries);

    try {
        while (running && c->connected) {
            uint8_t type = 0;
            std::string payload;

            if (!CipherTransport::receive_frame(c->fd, type, payload))
                break;

            ++c->stats.received_bytes;
            c->stats.received_bytes += payload.size();

            if (type == CipherTransport::TEXT) {
                std::string message = libraries.decode(payload);
                ++c->stats.received_messages;
                c->stats.received_chars += message.size();
                Neon::decode("CLIENT " + std::to_string(c->id) + ": " + message);
                Neon::print_message("CLIENT " +
                                    std::to_string(c->id), message);

                if (message == "/quit")
                    break;
            }
            else if (type == CipherTransport::FAST_TEXT) {
                ++c->stats.received_messages;
                c->stats.received_chars += payload.size();
                Neon::print_message("CLIENT " +
                                    std::to_string(c->id), payload);

                if (payload == "/quit")
                    break;
            }
            else if (type == CipherTransport::IMAGE) {
                std::filesystem::path dir =
                    "received_images/client_" + std::to_string(c->id);
                std::filesystem::create_directories(dir);
                std::string path =
                    ImageLibrary::unpack(payload, dir.string());
                Neon::info("Image received: " + path);
            }
            else if (type == CipherTransport::RAW_IMAGE) {
                std::filesystem::path dir =
                    "received_images/client_" + std::to_string(c->id);
                std::filesystem::create_directories(dir);
                std::string path =
                    ImageLibrary::unpack_raw(payload, dir.string());
                Neon::info("Raw image received: " + path);
            }
            else if (type >= CipherTransport::VIDEO_START &&
                     type <= CipherTransport::VIDEO_END) {
                std::string result = video_receiver.handle(type, payload);
                if (!result.empty())
                    Neon::info("Video received: " + result);
            }
            else {
                Neon::error("Unknown frame from client " +
                            std::to_string(c->id));
            }
        }
    }
    catch (const std::exception& e) {
        Neon::error("Client " + std::to_string(c->id) +
                    ": " + e.what());
    }

    disconnect_client(c);
    Neon::info("Client " + std::to_string(c->id) + " disconnected");
}

static std::shared_ptr<Client> find_client(int id) {
    std::lock_guard<std::mutex> lock(clients_mutex);

    for (auto& c : clients)
        if (c->id == id && c->connected)
            return c;

    return nullptr;
}

static void cleanup_clients() {
    std::vector<std::shared_ptr<Client>> dead;

    {
        std::lock_guard<std::mutex> lock(clients_mutex);

        for (auto it = clients.begin(); it != clients.end();) {
            if (!(*it)->connected) {
                dead.push_back(*it);
                it = clients.erase(it);
            } else {
                ++it;
            }
        }
    }

    for (auto& c : dead) {
        if (c->receiver.joinable())
            c->receiver.join();

        close(c->fd);
    }
}

static void transmit_image(

    const std::string& path,
    bool raw
) {

    if (!c->connected)

        : ImageLibrary::pack(path);

    auto type = raw
        ? CipherTransport::RAW_IMAGE
        : CipherTransport::IMAGE;

    CipherTransport::send_frame(c->fd, type, packet);

    ++c->stats.sent_messages;
    c->stats.sent_bytes += packet.size();
}

int main() {
    try {
        LibraryManager libraries("libraries");

        int server = socket(AF_INET, SOCK_STREAM, 0);
        if (server < 0)
            throw std::runtime_error("Socket creation failed");

        int opt = 1;
        setsockopt(server, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt));

        sockaddr_in address{};
        address.sin_family = AF_INET;
        address.sin_addr.s_addr = INADDR_ANY;
        address.sin_port = htons(8080);

        if (bind(server, reinterpret_cast<sockaddr*>(&address),
                 sizeof(address)) < 0)
            throw std::runtime_error("Bind failed");

        if (listen(server, 20) < 0)
            throw std::runtime_error("Listen failed");

        Neon::banner("SERVER");
        Neon::info("CipherAPI multi-client server started on port 8080");

        std::thread accept_thread([&]() {
            int next_id = 1;

            while (running) {
                pollfd pfd{server, POLLIN, 0};
                int ready = poll(&pfd, 1, 500);

                if (!running)
                    break;

                if (ready <= 0)
                    continue;

                sockaddr_in client_addr{};
                socklen_t len = sizeof(client_addr);

                int fd = accept(server,
                    reinterpret_cast<sockaddr*>(&client_addr), &len);

                if (fd < 0)
                    continue;

                auto c = std::make_shared<Client>();
                c->id = next_id++;
                c->fd = fd;

                c->receiver = std::thread(
                    receive_client, c, std::cref(libraries));

                {
                    std::lock_guard<std::mutex> lock(clients_mutex);
                    clients.push_back(c);
                }

                Neon::info("Client " + std::to_string(c->id) +
                           " connected from " +
                           std::string(inet_ntoa(client_addr.sin_addr)));
            }
        });

        int selected_id = -1;
        std::string line;

        Neon::info("Type /help for commands");

        while (running) {
            cleanup_clients();

            std::cout << "CipherAPI";
            if (selected_id != -1)
                std::cout << "[" << selected_id << "]";
            std::cout << "> " << std::flush;

            if (!std::getline(std::cin, line))
                break;

            if (line.empty())
                continue;

            std::istringstream input(line);
            std::string command;
            input >> command;

            try {
                if (command == "/help") {
                    std::cout
                        << "\n/help                 Show commands\n"
                        << "/clients              List clients\n"
                        << "/use ID               Select client\n"
                        << "/to ID MESSAGE        Direct message\n"
                        << "/broadcast MESSAGE    Send to everyone\n"
                        << "/stats                Selected client stats\n"
                        << "/fast                 Toggle fast mode\n"
                        << "/sendimg FILE         Send image\n"
                        << "/sendimg -f FILE      Send raw image\n"
                        << "/sendvid FILE         Send video\n"
                        << "/sendvid -f FILE      Send video with SHA-256\n"
                        << "/quit                 Disconnect selected client\n"
                        << "/shutdown             Stop server\n";
                }
                else if (command == "/clients") {
                    show_clients();
                }
                else if (command == "/use") {
                    int id;
                    if (!(input >> id)) {
                        Neon::error("Usage: /use ID");
                        continue;
                    }

                    if (find_client(id)) {
                        selected_id = id;
                        Neon::info("Selected client " + std::to_string(id));
                    } else {
                        Neon::error("Client not found");
                    }
                }
                else if (command == "/to") {
                    int id;
                    if (!(input >> id)) {
                        Neon::error("Usage: /to ID MESSAGE");
                        continue;
                    }

                    std::string message;
                    std::getline(input >> std::ws, message);

                    auto c = find_client(id);
                    if (!c || message.empty()) {
                        Neon::error("Client not found or message empty");
                        continue;
                    }

                    send_text(c, libraries, message);
                }
                else if (command == "/broadcast") {
                    std::string message;
                    std::getline(input >> std::ws, message);

                    if (message.empty())
                        continue;

                    std::vector<std::shared_ptr<Client>> snapshot;
                    {
                        std::lock_guard<std::mutex> lock(clients_mutex);
                        snapshot = clients;
                    }

                    for (auto& c : snapshot) {
                        if (!c->connected)
                            continue;

                        try {
                            send_text(c, libraries, message);
                        } catch (...) {
                            disconnect_client(c);
                        }
                    }
                }
                else if (command == "/stats") {
                    auto c = find_client(selected_id);
                    if (c)
                        Neon::show_stats(c->stats);
                    else
                        Neon::error("No active client selected");
                }
                else if (command == "/fast") {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("Select a client first");
                        continue;
                    }

                    c->fast = !c->fast;
                    Neon::info(std::string("Fast mode: ") +
                               (c->fast ? "ON" : "OFF"));
                }
                else if (command == "/sendimg" ||
                         command == "/sendvid") {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("Select a client first");
                        continue;
                    }

                    std::string option, path;
                    input >> option;

                    bool raw = option == "-f";
                    if (raw)
                        input >> path;
                    else
                        path = option;

                    if (path.empty()) {
                        Neon::error("File path required");
                        continue;
                    }

                    if (command == "/sendimg")
                        transmit_image(c, path, raw);
                    else
                        transmit_video(c, libraries, path, raw);
                }
                else if (command == "/quit") {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("No active client selected");
                        continue;
                    }

                    try {
                        send_text(c, libraries, "/quit");
                    } catch (...) {}

                    disconnect_client(c);
                    selected_id = -1;
                }
                else if (command == "/shutdown") {
                    running = false;
                    break;
                }
                else {
                    auto c = find_client(selected_id);
                    if (!c) {
                        Neon::error("Select a client using /use ID");
                        continue;
                    }

                    send_text(c, libraries, line);
                }
            }
            catch (const std::exception& e) {
                Neon::error(e.what());

                auto c = find_client(selected_id);
                if (c && !c->connected)
                    selected_id = -1;
            }
        }

        running = false;
        shutdown(server, SHUT_RDWR);

        {
            std::lock_guard<std::mutex> lock(clients_mutex);
            for (auto& c : clients)
                disconnect_client(c);
        }

        if (accept_thread.joinable())
            accept_thread.join();

        cleanup_clients();

        {
            std::lock_guard<std::mutex> lock(clients_mutex);
            for (auto& c : clients) {
                disconnect_client(c);
                if (c->receiver.joinable())
                    c->receiver.join();
                close(c->fd);
            }
            clients.clear();
        }

        close(server);
        Neon::info("CipherAPI server stopped");
        return 0;
    }
    catch (const std::exception& e) {
        std::cerr << "FATAL: " << e.what() << '\n';
        return 1;
    }
}

void put_u16(std::string& out, std::uint16_t v) {
    out.push_back(static_cast<char>(v >> 8));
    out.push_back(static_cast<char>(v));
}

void put_u32(std::string& out, std::uint32_t v) {
    for (int i = 3; i >= 0; --i)
        out.push_back(static_cast<char>(v >> (i * 8)));
}

void put_u64(std::string& out, std::uint64_t v) {
    for (int i = 7; i >= 0; --i)
        out.push_back(static_cast<char>(v >> (i * 8)));
}

std::uint16_t get_u16(const std::string& s, std::size_t p) {
    return (static_cast<std::uint16_t>(
                static_cast<unsigned char>(s[p])) << 8) |
           static_cast<unsigned char>(s[p + 1]);
}

std::uint32_t get_u32(const std::string& s, std::size_t p) {
    std::uint32_t v = 0;
    for (int i = 0; i < 4; ++i)
        v = (v << 8) | static_cast<unsigned char>(s[p + i]);
    return v;
}

std::uint64_t get_u64(const std::string& s, std::size_t p) {
    std::uint64_t v = 0;
    for (int i = 0; i < 8; ++i)
        v = (v << 8) | static_cast<unsigned char>(s[p + i]);
    return v;
}

// A deterministic generator: both ends produce the same permutation.
class SplitMix64 {
public:
    explicit SplitMix64(std::uint64_t seed) : state_(seed) {}

    std::uint64_t next() {
        std::uint64_t z = (state_ += 0x9e3779b97f4a7c15ULL);
        z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ULL;
        z = (z ^ (z >> 27)) * 0x94d049bb133111ebULL;
        return z ^ (z >> 31);
    }

private:
    std::uint64_t state_;
};

std::vector<std::uint32_t> make_permutation(
    std::size_t count, std::uint64_t seed) {

    std::vector<std::uint32_t> permutation(count);
    std::iota(permutation.begin(), permutation.end(), 0);

    SplitMix64 rng(seed);

    for (std::size_t i = count; i > 1; --i) {
        const std::size_t j =
            static_cast<std::size_t>(rng.next() % i);
        std::swap(permutation[i - 1], permutation[j]);
    }

    return permutation;
}

std::string bytes_to_bits(const std::string& bytes) {
    std::string bits;
    bits.reserve(bytes.size() * 8);

    for (unsigned char byte : bytes) {
        for (int bit = 7; bit >= 0; --bit)
            bits.push_back((byte & (1u << bit)) ? '1' : '0');
    }

    return bits;
}

std::string bits_to_bytes(const std::string& bits) {
    if (bits.size() % 8 != 0)
        throw std::runtime_error("Invalid reconstructed bit count");

    std::string bytes(bits.size() / 8, '\0');

    for (std::size_t i = 0; i < bits.size(); ++i) {
        if (bits[i] != '0' && bits[i] != '1')
            throw std::runtime_error("Invalid bit in shuffled data");

        if (bits[i] == '1')
            bytes[i / 8] |= static_cast<char>(1u << (7 - (i % 8)));
    }

    return bytes;
}

std::string shuffle_bits(
    const std::string& original, std::uint64_t seed) {

    const auto permutation =
        make_permutation(original.size(), seed);

    std::string shuffled(original.size(), '0');

    for (std::size_t i = 0; i < original.size(); ++i)
        shuffled[i] = original[permutation[i]];

    return shuffled;
}

std::string unshuffle_bits(
    const std::string& shuffled, std::uint64_t seed) {

    const auto permutation =
        make_permutation(shuffled.size(), seed);

    std::string original(shuffled.size(), '0');

    for (std::size_t i = 0; i < shuffled.size(); ++i)
        original[permutation[i]] = shuffled[i];

    return original;
}

void digest_update(EVP_MD_CTX* ctx, const std::string& bytes) {
    if (EVP_DigestUpdate(ctx, bytes.data(), bytes.size()) != 1)
        throw std::runtime_error("SHA-256 update failed");
}

std::string digest_final(EVP_MD_CTX* ctx) {
    std::string digest(HASH_SIZE, '\0');
    unsigned int length = 0;

    if (EVP_DigestFinal_ex(
            ctx,
            reinterpret_cast<unsigned char*>(digest.data()),
            &length) != 1 || length != HASH_SIZE) {
        throw std::runtime_error("SHA-256 finalization failed");
    }

    return digest;
}

std::string safe_output_name(const std::string& name) {
    if (name.empty() || name.size() > 255 ||
        name == "." || name == ".." ||
        fs::path(name).filename().string() != name ||
        name.find('/') != std::string::npos ||
        name.find('\\') != std::string::npos) {
        throw std::runtime_error("Unsafe video filename");
    }

    return name;
}

TransferStats send_file(
    int fd,
    const LibraryManager& libraries,
    const std::string& path,
    bool use_sha256,
    bool fast_mode) {

    (void)fast_mode; // Reserved for future video fast-mode support.
    const fs::path source(path);
    const std::string filename = source.filename().string();

    if (filename.empty() || filename.size() > 255)
        throw std::runtime_error("Invalid video filename");

    std::error_code ec;
    const std::uint64_t file_size = fs::file_size(source, ec);

    if (ec)
        throw std::runtime_error("Cannot read video file size: " + ec.message());

    if (file_size == 0)
        throw std::runtime_error("Video file is empty");

    std::ifstream input(source, std::ios::binary);
    if (!input)
        throw std::runtime_error("Cannot open video file");

    EVP_MD_CTX* digest = nullptr;

    if (use_sha256) {
        digest = EVP_MD_CTX_new();
        if (!digest)
            throw std::runtime_error("Cannot allocate SHA-256 context");

        if (EVP_DigestInit_ex(digest, EVP_sha256(), nullptr) != 1) {
            EVP_MD_CTX_free(digest);
            throw std::runtime_error("Cannot initialize SHA-256");
        }
    }

    TransferStats stats;
    stats.original_bytes = file_size;

    try {
        // VIDEO_START: version, flags, filename length, file size, filename.
        std::string start;
        start.push_back(static_cast<char>(VERSION));
        start.push_back(static_cast<char>(use_sha256 ? 1 : 0));
        put_u16(start, static_cast<std::uint16_t>(filename.size()));
        put_u64(start, file_size);
        start += filename;

        CipherTransport::send_frame(
            fd, CipherTransport::VIDEO_START, start);
        stats.wire_bytes += start.size();

        std::uint64_t remaining = file_size;
        std::uint32_t sequence = 0;

        while (remaining > 0) {
            const std::size_t amount = static_cast<std::size_t>(
                std::min<std::uint64_t>(CHUNK_SIZE, remaining));

            std::string bytes(amount, '\0');
            input.read(bytes.data(), static_cast<std::streamsize>(amount));

            if (input.gcount() != static_cast<std::streamsize>(amount))
                throw std::runtime_error("Video file changed or read failed");

            if (use_sha256)
                digest_update(digest, bytes);

            std::uint64_t seed = 0;
            if (RAND_bytes(
                    reinterpret_cast<unsigned char*>(&seed),
                    sizeof(seed)) != 1) {
                throw std::runtime_error("Random seed generation failed");
            }

            const std::string bits = bytes_to_bits(bytes);
            const std::string shuffled = shuffle_bits(bits, seed);

            // The existing ten libraries encode the shuffled 0/1 characters.
            const std::string encoded = libraries.encode(shuffled);

            // VIDEO_DATA: sequence, seed, original byte count, encoded bits.
            std::string packet;
            packet.reserve(CHUNK_HEADER_SIZE + encoded.size());

            put_u32(packet, sequence);
            put_u64(packet, seed);
            put_u16(packet, static_cast<std::uint16_t>(amount));
            packet += encoded;

            CipherTransport::send_frame(
                fd, CipherTransport::VIDEO_DATA, packet);

            stats.wire_bytes += packet.size();
            ++stats.chunks;
            ++sequence;
            remaining -= amount;
        }

        if (input.peek() != std::char_traits<char>::eof())
            throw std::runtime_error("Video file grew during transfer");

        // VIDEO_END: total original bytes, chunk count, optional digest.
        std::string end;
        put_u64(end, file_size);
        put_u32(end, stats.chunks);

        if (use_sha256)
            end += digest_final(digest);

        CipherTransport::send_frame(
            fd, CipherTransport::VIDEO_END, end);
        stats.wire_bytes += end.size();

        if (digest)
            EVP_MD_CTX_free(digest);

        return stats;
    } catch (...) {
        if (digest)
            EVP_MD_CTX_free(digest);
        throw;
    }
}

Receiver::Receiver(const LibraryManager& libraries)
    : libraries_(libraries), state_(new State()) {}

Receiver::~Receiver() {
    abort_transfer();
    delete state_;
}

void Receiver::abort_transfer() noexcept {
    if (state_ && state_->file.is_open())
        state_->file.close();

    if (!temporary_path_.empty()) {
        std::error_code ec;
        fs::remove(temporary_path_, ec);
    }

    active_ = false;
    expected_size_ = 0;
    received_size_ = 0;
    expected_sequence_ = 0;
    use_sha256_ = false;
    final_path_.clear();
    temporary_path_.clear();

    if (state_ && state_->digest) {
        EVP_MD_CTX_free(state_->digest);
        state_->digest = nullptr;
    }
}

std::string Receiver::handle(
    std::uint8_t type, const std::string& payload) {

    if (type == CipherTransport::VIDEO_START) {
        if (active_)
            throw std::runtime_error("A video transfer is already active");

        if (payload.size() < 12)
            throw std::runtime_error("Invalid VIDEO_START header");

        if (static_cast<std::uint8_t>(payload[0]) != VERSION)
            throw std::runtime_error("Unsupported CVS version");

        const std::uint8_t flags =
            static_cast<std::uint8_t>(payload[1]);

        if (flags > 1)
            throw std::runtime_error("Invalid video checksum mode");

        const std::uint16_t name_length = get_u16(payload, 2);
        expected_size_ = get_u64(payload, 4);

        if (name_length == 0 || name_length > 255 ||
            payload.size() != 12 + name_length ||
            expected_size_ == 0) {
            throw std::runtime_error("Invalid video metadata");
        }

        const std::string filename =
            safe_output_name(payload.substr(12, name_length));

        use_sha256_ = flags == 1;

        if (use_sha256_) {
            state_->digest = EVP_MD_CTX_new();
            if (!state_->digest ||
                EVP_DigestInit_ex(
                    state_->digest, EVP_sha256(), nullptr) != 1) {
                throw std::runtime_error("Cannot initialize receive SHA-256");
            }
        }

        std::string candidate = "received_" + filename;
        unsigned int suffix = 1;

        while (fs::exists(candidate) || fs::exists(candidate + ".part")) {
            candidate = "received_" + filename + "." +
                        std::to_string(suffix++);
        }

        final_path_ = candidate;
        temporary_path_ = final_path_ + ".part";

        state_->file.open(
            temporary_path_, std::ios::binary | std::ios::out);

        if (!state_->file)
            throw std::runtime_error("Cannot create temporary video file");

        received_size_ = 0;
        expected_sequence_ = 0;
        active_ = true;
        return {};
    }

    if (!active_)
        throw std::runtime_error("Video data received without VIDEO_START");

    if (type == CipherTransport::VIDEO_DATA) {
        if (payload.size() < CHUNK_HEADER_SIZE)
            throw std::runtime_error("Invalid VIDEO_DATA header");

        const std::uint32_t sequence = get_u32(payload, 0);
        const std::uint64_t seed = get_u64(payload, 4);
        const std::uint16_t raw_size = get_u16(payload, 12);

        if (sequence != expected_sequence_)
            throw std::runtime_error("Video chunk sequence mismatch");

        if (raw_size == 0 || raw_size > CHUNK_SIZE ||
            raw_size > expected_size_ - received_size_) {
            throw std::runtime_error("Invalid video chunk size");
        }

        const std::string encoded =
            payload.substr(CHUNK_HEADER_SIZE);

        if (encoded.size() !=
            static_cast<std::size_t>(raw_size) * 8 * 64) {
            throw std::runtime_error("Invalid encoded video chunk length");
        }

        const std::string shuffled = libraries_.decode(encoded);

        if (shuffled.size() != static_cast<std::size_t>(raw_size) * 8)
            throw std::runtime_error("Decoded bit count mismatch");

        const std::string bits = unshuffle_bits(shuffled, seed);
        const std::string bytes = bits_to_bytes(bits);

        if (bytes.size() != raw_size)
            throw std::runtime_error("Reconstructed chunk size mismatch");

        state_->file.write(
            bytes.data(), static_cast<std::streamsize>(bytes.size()));

        if (!state_->file)
            throw std::runtime_error("Video disk write failed");

        if (use_sha256_)
            digest_update(state_->digest, bytes);

        received_size_ += bytes.size();
        ++expected_sequence_;
        return {};
    }

    if (type == CipherTransport::VIDEO_END) {
        const std::size_t expected_end_size =
            use_sha256_ ? 12 + HASH_SIZE : 12;

        if (payload.size() != expected_end_size)
            throw std::runtime_error("Invalid VIDEO_END packet");

        const std::uint64_t final_size = get_u64(payload, 0);
        const std::uint32_t final_chunks = get_u32(payload, 8);

        const std::uint64_t expected_chunks =
            (expected_size_ + CHUNK_SIZE - 1) / CHUNK_SIZE;

        if (final_size != expected_size_ ||
            received_size_ != expected_size_ ||
            final_chunks != expected_sequence_ ||
            final_chunks != expected_chunks) {
            throw std::runtime_error("Video transfer size or chunk mismatch");
        }

        if (use_sha256_) {
            const std::string actual = digest_final(state_->digest);
            const std::string claimed = payload.substr(12, HASH_SIZE);

            if (CRYPTO_memcmp(
                    actual.data(), claimed.data(), HASH_SIZE) != 0) {
                throw std::runtime_error("Video SHA-256 verification failed");
            }
        }

        state_->file.flush();
        if (!state_->file)
            throw std::runtime_error("Video file flush failed");

        state_->file.close();

        std::error_code ec;
        fs::rename(temporary_path_, final_path_, ec);

        if (ec)
            throw std::runtime_error("Cannot finalize video file: " +
                                     ec.message());

        const std::string saved = final_path_;

        active_ = false;
        expected_size_ = 0;
        received_size_ = 0;
        expected_sequence_ = 0;
        use_sha256_ = false;
        final_path_.clear();
        temporary_path_.clear();

        if (state_->digest) {
            EVP_MD_CTX_free(state_->digest);
            state_->digest = nullptr;
        }

        return saved;
    }

    throw std::runtime_error("Unexpected frame during video transfer");
}
